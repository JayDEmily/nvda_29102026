from product_proof_claims import load_product_proof_claim

def clone(name):
 path=OUT/(name+'-v2')
 subprocess.run(['git','clone','--quiet','--no-hardlinks',str(ROOT),str(path)],check=True)
 subprocess.run(['git','checkout','--quiet','--detach',CANDIDATE],cwd=path,check=True)
 return path
def commit(path):
 subprocess.run(['git','add','-A'],cwd=path,check=True)
 subprocess.run(['git','-c','user.name=Independent G53 review','-c','user.email=review@example.invalid','commit','-qm','Disposable adversarial control'],cwd=path,check=True)
def run(path):
 env=dict(os.environ,PYTHONPATH=str(path/'tools')+os.pathsep+str(path/'src'))
 p=subprocess.run([sys.executable,str(path/'tools/run_product_proof_gate.py'),'--aggregate',CLAIM],cwd=path,env=env,text=True,capture_output=True)
 print(path.name,'exit',p.returncode,p.stdout.strip(),p.stderr.strip(),flush=True)
 assert p.returncode!=0
 return p
