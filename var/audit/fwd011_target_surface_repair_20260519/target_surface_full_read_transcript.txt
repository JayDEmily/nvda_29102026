    23	class _UniqueKeyLoader(yaml.SafeLoader):
    24	    """YAML loader that rejects duplicate mapping keys before they collapse."""
    25	
    26	
    27	def _construct_mapping_without_duplicates(loader: yaml.Loader, node: yaml.Node, deep: bool = False):
    28	    mapping = {}
    29	    for key_node, value_node in node.value:
    30	        key = loader.construct_object(key_node, deep=deep)
    31	        if key in mapping:
    32	            raise ConstructorError(
    33	                "while constructing a mapping",
    34	                node.start_mark,
    35	                f"duplicate YAML key: {key}",
    36	                key_node.start_mark,
    37	            )
    38	        mapping[key] = loader.construct_object(value_node, deep=deep)
    39	    return mapping

     1	"""Pure playbook admissibility compilation."""
     2	
     3	from __future__ import annotations
     4	
     5	from nvda_desk.config.playbook_registry import (
     6	    PlaybookRegistryDocument,
     7	    build_playbook_metadata_receipts,
     8	)
     9	from nvda_desk.contracts.eligibility import (
    10	    EligibilityAdmissibilitySurface,
    11	    PlaybookEligibilityOutput,
    12	)
    13	from nvda_desk.contracts.posture import PermissionState
    14	
    15	
    16	def _unique_in_order(values: list[str] | tuple[str, ...]) -> list[str]:
    17	    seen: set[str] = set()
    18	    result: list[str] = []
    19	    for value in values:
    20	        if value not in seen:
    21	            seen.add(value)
    22	            result.append(value)
    23	    return result
    24	
    25	
    26	def compile_playbook_eligibility(
    27	    *,
    28	    permission_state: PermissionState,
    29	    candidate_family_ids: list[str] | None = None,
    30	    registry: PlaybookRegistryDocument | None = None,
    31	) -> PlaybookEligibilityOutput:
    32	    """Return one bounded eligibility surface from permission state and playbook inputs."""
    33	
    34	    families = _unique_in_order(candidate_family_ids or [])
    35	    no_trade_reasons: list[str] = []
    36	    registry_id = registry.registry_id if registry is not None else None
    37	
    38	    if registry is not None and not families:
    39	        families = _unique_in_order(
    40	            [playbook.setup_family for playbook in registry.enabled_playbooks()]
    41	        )
    42	
    43	    if permission_state == PermissionState.BLOCK:
    44	        no_trade_reasons.append("permission_block")
    45	        return PlaybookEligibilityOutput(
    46	            admissibility_surface=EligibilityAdmissibilitySurface(
    47	                permission_state=permission_state,
    48	                no_trade_reasons=no_trade_reasons,
    49	            ),
    50	            active_family_ids=[],
    51	            active_setup_variant_ids=[],
    52	            registry_id=registry_id,
    53	            playbook_metadata_receipts=[],
    54	        )
    55	
    56	    active_setup_variant_ids: list[str]
    57	    if registry is None:
    58	        active_setup_variant_ids = [f"{family}.baseline" for family in families]
    59	        receipts = []
    60	    else:
    61	        receipts = build_playbook_metadata_receipts(registry, setup_family_ids=families)
    62	        active_setup_variant_ids = _unique_in_order(
    63	            [receipt.setup_variant_id for receipt in receipts]
    64	        )
    65	        families = _unique_in_order([receipt.setup_family for receipt in receipts])
    66	
    67	    return PlaybookEligibilityOutput(
    68	        admissibility_surface=EligibilityAdmissibilitySurface(
    69	            permission_state=permission_state,
    70	            no_trade_reasons=no_trade_reasons,
    71	        ),
    72	        active_family_ids=families,
    73	        active_setup_variant_ids=active_setup_variant_ids,
    74	        registry_id=registry_id,
    75	        playbook_metadata_receipts=receipts,
    76	    )
