class _DetachedMutableModel(BaseModel):
    """Base model that detaches caller-owned mutable field values."""

    @field_validator("*", mode="before")
    @classmethod
    def _detach_mutable_values(cls, value):
        return deepcopy(value)

def _payload_keys(value: Any):
    if isinstance(value, dict):
        for key, child in value.items():
            yield str(key)
            yield from _payload_keys(child)
    elif isinstance(value, (list, tuple, set)):
        for child in value:
            yield from _payload_keys(child)


def _reject_live_inventory_transform_required_donor_payload(value: Any):
    observed = set(_payload_keys(value))
    forbidden = observed & _LIVE_INVENTORY_TRANSFORM_REQUIRED_DONOR_PAYLOAD_KEYS
    if forbidden:
        joined = ", ".join(sorted(forbidden))
        raise ValueError(
            "live inventory source_payload cannot carry transform-required donor "
            f"payload keys: {joined}"
        )
    return value

        "broker_execution_record",

        "ExecutionRecord",
        "fill_price_usd",

def _reject_placement_handoff_broker_execution_payload(value: Any):
    observed = set(_payload_keys(value))
    execution_record_keys = observed & _PLACEMENT_HANDOFF_EXECUTION_RECORD_DONOR_KEYS
    if execution_record_keys:
        joined = ", ".join(sorted(execution_record_keys))
        raise ValueError(
            "placement handoff order_intent_preview cannot carry execution record "
            f"donor keys: {joined}"
        )
    forbidden = observed & _PLACEMENT_HANDOFF_BROKER_EXECUTION_KEYS
    if forbidden:
        joined = ", ".join(sorted(forbidden))
        raise ValueError(
            "placement handoff order_intent_preview cannot carry broker execution "
            f"keys: {joined}"
        )
    return value

"""Build bounded live-inventory snapshots."""

from __future__ import annotations

from collections.abc import Iterable, Mapping
from copy import deepcopy
from typing import Any

from nvda_desk.contracts.downstream import (
    DownstreamLiveInventorySnapshot,
    InventoryExposureState,
    InventoryStateReceipt,
    LiveCapitalState,
    LivePositionState,
)
from nvda_desk.persistence.receipt_ledger import ReceiptLedger
from nvda_desk.repositories.capital_state_repository import CapitalStateRepository
from nvda_desk.repositories.position_repository import PositionRepository


class DownstreamLiveInventoryService:
    """Create truthful live-inventory snapshots without injecting policy."""

    def __init__(
        self,
        *,
        position_repository: PositionRepository | None = None,
        capital_state_repository: CapitalStateRepository | None = None,
        receipt_ledger: ReceiptLedger | None = None,
    ):
        self._position_repository = position_repository
        self._capital_state_repository = capital_state_repository
        self._receipt_ledger = receipt_ledger

    def build_live_snapshot(
        self,
        live_ids: list[str] | None = None,
        *,
        inventory_id: str = "live-inventory",
        capital_snapshot_id: str | None = None,
        receipt_id: str | None = None,
    ) -> DownstreamLiveInventorySnapshot:
        """Build one inventory snapshot from repo-native state without policy decisions."""

        requested_ids = _normalise_id_list(live_ids)
        position_states = self._read_position_states(requested_ids)
        capital_state = self._read_capital_state(capital_snapshot_id)
        exposure_state = _build_exposure_state(position_states)
        receipt = None
        if receipt_id is not None:
            receipt = InventoryStateReceipt(
                receipt_id=_normalise_token(receipt_id, "receipt_id"),
                inventory_id=_normalise_token(inventory_id, "inventory_id"),
                position_ids=[state.position_id for state in position_states],
                capital_snapshot_id=capital_state.snapshot_id if capital_state else None,
            )
            if self._receipt_ledger is not None:
                self._receipt_ledger.append_receipt(
                    receipt.receipt_id,
                    "live_inventory_snapshot",
                    receipt.model_dump(),
                )
        return DownstreamLiveInventorySnapshot(
            inventory_id=_normalise_token(inventory_id, "inventory_id"),
            live_ids=[state.position_id for state in position_states] or requested_ids,
            position_states=position_states,
            capital_state=capital_state,
            exposure_state=exposure_state,
            inventory_receipt=receipt,
        )

    def reconcile_fill_evidence(
        self,
        fill_evidence: Iterable[Mapping[str, Any]],
        *,
        reconciliation_id: str,
        inventory_id: str = "live-inventory",
        capital_snapshot_id: str | None = None,
        receipt_id: str | None = None,
    ) -> DownstreamLiveInventorySnapshot:
        """Reconcile executed fill evidence into repo-native live-inventory state."""

        if self._position_repository is None:
            raise ValueError("position_repository is required for fill reconciliation")

        normalised_reconciliation_id = _normalise_token(
            reconciliation_id, "reconciliation_id"
        )
        normalised_events = [_normalise_fill_evidence_event(event) for event in fill_evidence]
        if not normalised_events:
            raise ValueError("fill_evidence must contain at least one event")
        _assert_unique_fill_evidence_ids(normalised_events)

        grouped_payloads: dict[str, dict[str, Any]] = {}
        for event in normalised_events:
            position_id = event["position_id"]
            payload = grouped_payloads.get(position_id)
            if payload is None:
                existing = self._position_repository.read_position(position_id)
                payload = deepcopy(existing) if existing is not None else {}
                payload.setdefault("fill_evidence_ids", [])
                payload.setdefault("fill_evidence", [])
                grouped_payloads[position_id] = payload

            existing_fill_ids = _normalised_existing_fill_ids(payload.get("fill_evidence_ids"))
            if event["fill_evidence_id"] in existing_fill_ids:
                raise ValueError(f"duplicate fill_evidence_id: {event['fill_evidence_id']}")

            if event.get("symbol") and not payload.get("symbol"):
                payload["symbol"] = event["symbol"]

            declared_delta = _optional_float(event.get("declared_exposure_delta"))
            if declared_delta is not None:
                current_declared = _optional_float(payload.get("declared_exposure")) or 0.0
                payload["declared_exposure"] = current_declared + declared_delta

            payload["fill_evidence_ids"].append(event["fill_evidence_id"])
            payload["fill_evidence"].append(deepcopy(event))
            payload["inventory_reconciliation_id"] = normalised_reconciliation_id

        for position_id, payload in grouped_payloads.items():
            if self._position_repository.read_position(position_id) is None:
                self._position_repository.write_position(position_id, payload)
            else:
                self._position_repository.replace_position(position_id, payload)

        reconciled_position_ids = list(grouped_payloads)
        snapshot = self.build_live_snapshot(
            live_ids=reconciled_position_ids,
            inventory_id=inventory_id,
            capital_snapshot_id=capital_snapshot_id,
        )
        normalised_receipt_id = receipt_id
        receipt = None
        if normalised_receipt_id is not None:
            receipt = InventoryStateReceipt(
                receipt_id=_normalise_token(normalised_receipt_id, "receipt_id"),
                inventory_id=_normalise_token(inventory_id, "inventory_id"),
                position_ids=reconciled_position_ids,
                capital_snapshot_id=snapshot.capital_state.snapshot_id
                if snapshot.capital_state
                else None,
            )
            if self._receipt_ledger is not None:
                receipt_payload = receipt.model_dump()
                receipt_payload.update(
                    {
                        "reconciliation_id": normalised_reconciliation_id,
                        "fill_evidence_ids": [
                            event["fill_evidence_id"] for event in normalised_events
                        ],
                    }
                )
                self._receipt_ledger.append_receipt(
                    receipt.receipt_id,
                    "live_inventory_reconciliation",
                    receipt_payload,
                )
        return snapshot.model_copy(update={"inventory_receipt": receipt})

    def _read_position_states(self, requested_ids: list[str]) -> list[LivePositionState]:
        if self._position_repository is None:
            return [LivePositionState(position_id=position_id) for position_id in requested_ids]

        if requested_ids:
            records = [
                self._position_repository.read_position_record(position_id)
                for position_id in requested_ids
            ]
            records = [record for record in records if record is not None]
        else:
            records = self._position_repository.list_position_records()

        return [
            LivePositionState(
                position_id=str(record["position_id"]),
                source_payload=deepcopy(dict(record["position"])),
            )
            for record in records
        ]

    def _read_capital_state(self, snapshot_id: str | None) -> LiveCapitalState | None:
        if self._capital_state_repository is None:
            return None
        if snapshot_id is None:
            records = self._capital_state_repository.list_snapshot_records()
            if not records:
                return None
            record = records[-1]
        else:
            record = self._capital_state_repository.read_snapshot_record(snapshot_id)
            if record is None:
                return None
        return LiveCapitalState(
            snapshot_id=str(record["snapshot_id"]),
            source_payload=deepcopy(dict(record["snapshot"])),
        )


_FILL_EVIDENCE_FORBIDDEN_KEYS = frozenset(
    {
        "broker_order_id",
        "order_submitted",
        "execution_allowed",
        "capital_deployment_receipt_id",
        "opening_authorisation",
        "placement_handoff",
    }
)


def _normalise_fill_evidence_event(event: Mapping[str, Any]) -> dict[str, Any]:
    payload = deepcopy(dict(event))
    forbidden = set(payload) & _FILL_EVIDENCE_FORBIDDEN_KEYS
    if forbidden:
        joined = ", ".join(sorted(forbidden))
        raise ValueError(f"fill evidence cannot carry placement authority keys: {joined}")
    payload["fill_evidence_id"] = _normalise_token(
        payload.get("fill_evidence_id", ""), "fill_evidence_id"
    )
    payload["position_id"] = _normalise_token(payload.get("position_id", ""), "position_id")
    if "symbol" in payload:
        symbol = _optional_text(payload["symbol"])
        if symbol is not None:
            payload["symbol"] = symbol
        else:
            payload.pop("symbol")
    declared_delta = _optional_float(payload.get("declared_exposure_delta"))
    if declared_delta is not None:
        payload["declared_exposure_delta"] = declared_delta
    return payload


def _assert_unique_fill_evidence_ids(events: list[dict[str, Any]]) -> None:
    observed: set[str] = set()
    for event in events:
        evidence_id = event["fill_evidence_id"]
        if evidence_id in observed:
            raise ValueError(f"duplicate fill_evidence_id: {evidence_id}")
        observed.add(evidence_id)


def _normalised_existing_fill_ids(values: Any) -> set[str]:
    if values is None:
        return set()
    if not isinstance(values, list):
        raise ValueError("fill_evidence_ids must be a list")
    return {_normalise_token(value, "fill_evidence_id") for value in values}


def _build_exposure_state(position_states: Iterable[LivePositionState]) -> InventoryExposureState:
    ids: list[str] = []
    exposures: dict[str, float] = {}
    for state in position_states:
        ids.append(state.position_id)
        symbol = _optional_text(state.source_payload.get("symbol"))
        declared_exposure = _optional_float(state.source_payload.get("declared_exposure"))
        if symbol and declared_exposure is not None:
            exposures[symbol] = exposures.get(symbol, 0.0) + declared_exposure
    return InventoryExposureState(
        position_count=len(ids),
        open_position_ids=ids,
        declared_exposure_by_symbol=exposures,
    )


def _normalise_id_list(values: list[str] | None) -> list[str]:
    if values is None:
        return []
    return [_normalise_token(value, "live_id") for value in values]


def _normalise_token(value: str, field_name: str) -> str:
    token = str(value).strip()
    if not token:
        raise ValueError(f"{field_name} must be non-empty")
    return token


def _optional_text(value: Any) -> str | None:
    if value is None:
        return None
    text = str(value).strip()
    return text or None


def _optional_float(value: Any) -> float | None:
    if value is None:
        return None
    if isinstance(value, bool):
        return None
    try:
        return float(value)
    except (TypeError, ValueError):
        return None

"""Persist canonical position snapshots in one bounded in-memory store."""

from __future__ import annotations

from copy import deepcopy
from typing import Any, Mapping


class PositionRepository:
    """Deterministic in-memory position repository for the rebuild phase."""

    def __init__(self):
        self._positions: dict[str, dict[str, Any]] = {}

    def write_position(self, position_id: str, position: Mapping[str, Any]) -> None:
        """Store one canonical position snapshot by identifier."""

        normalised_id = str(position_id).strip()
        if not normalised_id:
            raise ValueError("position_id must be non-empty")
        if normalised_id in self._positions:
            raise ValueError(f"duplicate position_id: {normalised_id}")
        self._positions[normalised_id] = deepcopy(dict(position))

    def read_position(self, position_id: str) -> dict[str, Any] | None:
        """Return one stored canonical position snapshot, if present."""

        record = self.read_position_record(position_id)
        return record["position"] if record is not None else None

    def replace_position(self, position_id: str, position: Mapping[str, Any]) -> None:
        """Replace one existing canonical position snapshot by identifier."""

        normalised_id = str(position_id).strip()
        if not normalised_id:
            raise ValueError("position_id must be non-empty")
        if normalised_id not in self._positions:
            raise KeyError(f"unknown position_id: {normalised_id}")
        self._positions[normalised_id] = deepcopy(dict(position))

    def read_position_record(self, position_id: str) -> dict[str, Any] | None:
        """Return one stored position with its repository identifier."""

        normalised_id = str(position_id).strip()
        position = self._positions.get(normalised_id)
        if position is None:
            return None
        return {"position_id": normalised_id, "position": deepcopy(position)}

    def list_positions(self) -> list[dict[str, Any]]:
        """Return stored canonical position snapshots in insertion order."""

        return deepcopy(list(self._positions.values()))

    def list_position_records(self) -> list[dict[str, Any]]:
        """Return stored positions with repository identifiers in insertion order."""

        return [
            {"position_id": position_id, "position": deepcopy(position)}

"""Persist canonical capital-state snapshots in one bounded in-memory store."""

from __future__ import annotations

from copy import deepcopy
from typing import Any, Mapping


class CapitalStateRepository:
    """Deterministic in-memory capital-state repository for the rebuild phase."""

    def __init__(self):
        self._snapshots: dict[str, dict[str, Any]] = {}

    def write_snapshot(self, snapshot_id: str, snapshot: Mapping[str, Any]) -> None:
        """Store one capital-state snapshot by identifier."""

        normalised_id = str(snapshot_id).strip()
        if not normalised_id:
            raise ValueError("snapshot_id must be non-empty")
        if normalised_id in self._snapshots:
            raise ValueError(f"duplicate snapshot_id: {normalised_id}")
        self._snapshots[normalised_id] = deepcopy(dict(snapshot))

    def read_snapshot(self, snapshot_id: str) -> dict[str, Any] | None:
        """Return one stored capital-state snapshot, if present."""

        record = self.read_snapshot_record(snapshot_id)
        return record["snapshot"] if record is not None else None

    def read_snapshot_record(self, snapshot_id: str) -> dict[str, Any] | None:
        """Return one stored capital-state snapshot with its repository identifier."""

        normalised_id = str(snapshot_id).strip()
        snapshot = self._snapshots.get(normalised_id)
        if snapshot is None:
            return None
        return {"snapshot_id": normalised_id, "snapshot": deepcopy(snapshot)}

    def list_snapshots(self) -> list[dict[str, Any]]:
        """Return stored capital-state snapshots in insertion order."""

        return deepcopy(list(self._snapshots.values()))

    def list_snapshot_records(self) -> list[dict[str, Any]]:
        """Return stored capital-state snapshots with identifiers in insertion order."""

        return [
            {"snapshot_id": snapshot_id, "snapshot": deepcopy(snapshot)}
            for snapshot_id, snapshot in self._snapshots.items()
        ]

"""Durable receipt-ledger primitives for bounded audit and checkpoint truth."""

from __future__ import annotations

import json
from copy import deepcopy
from typing import Any, Mapping


class ReceiptLedger:
    """Append-only receipt ledger with duplicate-id protection and copy-on-read."""

    def __init__(self):
        self._receipts: list[dict[str, Any]] = []
        self._receipt_ids: set[str] = set()

    def append_receipt(
        self, receipt_id: str, receipt_kind: str, payload: Mapping[str, Any]
    ) -> None:
        """Append one JSON-serialisable receipt payload by stable identifier."""

        normalised_id = _normalise_token(receipt_id, "receipt_id")
        normalised_kind = _normalise_token(receipt_kind, "receipt_kind")
        if normalised_id in self._receipt_ids:
            raise ValueError(f"duplicate receipt_id: {normalised_id}")

        stored_payload = deepcopy(dict(payload))
        _assert_json_serialisable(stored_payload)
        self._receipt_ids.add(normalised_id)
        self._receipts.append(
            {
                "receipt_id": normalised_id,
                "receipt_kind": normalised_kind,
                "payload": stored_payload,
            }
        )

    def read_receipt(self, receipt_id: str) -> dict[str, Any] | None:
        """Return one stored receipt by identifier, if present."""

        normalised_id = _normalise_token(receipt_id, "receipt_id")
        for receipt in self._receipts:
            if receipt["receipt_id"] == normalised_id:
                return deepcopy(receipt)
        return None

    def list_receipts(self) -> list[dict[str, Any]]:
        """Return stored receipts in append order."""

        return deepcopy(self._receipts)

    def list_receipts_by_kind(self, receipt_kind: str) -> list[dict[str, Any]]:
        """Return stored receipts of one kind in append order."""

        normalised_kind = _normalise_token(receipt_kind, "receipt_kind")
        return deepcopy(
            [
                receipt
                for receipt in self._receipts
                if receipt["receipt_kind"] == normalised_kind
            ]
        )


def _normalise_token(value: str, field_name: str) -> str:
    token = str(value).strip()
    if not token:
        raise ValueError(f"{field_name} must be non-empty")
    return token


def _assert_json_serialisable(payload: Mapping[str, Any]) -> None:
    try:
        json.dumps(payload, sort_keys=True)
    except TypeError as exc:
        raise TypeError("receipt payload must be JSON-serialisable") from exc

class DeskRuntimeOrchestrator:
    """Own one deterministic stateless runtime pass and explicit serial stage order."""

    stage_order: tuple[str, ...] = SERIAL_SPINE_STAGE_ORDER

    _stage_dependencies: dict[str, tuple[str, ...]] = validate_serial_stage_dependencies(
        SERIAL_SPINE_STAGE_ORDER, SERIAL_STAGE_DEPENDENCIES
    )

    _stage_packet_metadata: dict[str, dict[str, str]] = {
        "temporal": {
            "module_id": "temporal_context",
            "grammar_role": "temporal_context",
            "dataschema": "TemporalContextOutput",
        },
        "regime": {
            "module_id": "market_regime_context",
            "grammar_role": "market_regime_context",
            "dataschema": "MarketRegimeContextOutput",
        },
        "options_flow": {
            "module_id": "options_flow_context",
            "grammar_role": "options_flow_context",
            "dataschema": "OptionsFlowContextOutput",
        },
        "posture": {
            "module_id": "posture_risk_permission",
            "grammar_role": "posture_risk_permission",
            "dataschema": "PostureRiskOutput",
        },
        "eligibility": {
            "module_id": "playbook_eligibility",
            "grammar_role": "playbook_eligibility",
            "dataschema": "PlaybookEligibilityOutput",
        },
        "execution": {
            "module_id": "expression_execution",
            "grammar_role": "expression_execution",
            "dataschema": "ExecutionExpressionOutput",
        },
    }

    def stage_sequence(self) -> tuple[str, ...]:
        """Return the governed six-stage serial-runtime order for one deterministic pass."""

        return self.stage_order

    def run_once(
        self,
        stage_inputs: Mapping[str, Any] | StageInputBundle,
        *,
        live_inventory_service: Any | None = None,
        live_inventory_request: Mapping[str, Any] | None = None,
        live_inventory_snapshot: DownstreamLiveInventorySnapshot | Mapping[str, Any] | None = None,
    ) -> dict[str, Any]:
        """Return one bounded runtime pass from the admitted stage-input boundary."""

        bundle = _coerce_admitted_bundle(stage_inputs)
        if bundle is None:
            normalised_stage_inputs = _normalise_stage_inputs(stage_inputs)
            executed = [name for name in self.stage_order if name in normalised_stage_inputs]
            receipt = {
                "stage_order": list(self.stage_order),
                "executed_stages": executed,
                "missing_stages": [
                    name for name in self.stage_order if name not in normalised_stage_inputs
                ],
            }
            return _attach_runtime_live_inventory(
                receipt,
                live_inventory_service=live_inventory_service,
                live_inventory_request=live_inventory_request,
                live_inventory_snapshot=live_inventory_snapshot,
            )
        receipt = self._run_serial_bundle(bundle)
        return _attach_runtime_live_inventory(
            receipt,
            live_inventory_service=live_inventory_service,
            live_inventory_request=live_inventory_request,
            live_inventory_snapshot=live_inventory_snapshot,
        )

    def _run_serial_bundle(self, stage_input_bundle: StageInputBundle) -> dict[str, Any]:
        source_time_utc = _require_source_time_utc(stage_input_bundle)
        token = _time_token(source_time_utc)
        traceid = f"trace::serial_runtime::{token}"
        runid = f"run::serial_runtime::{token}"
        scenarioid = "live"
        inputfingerprint = _input_fingerprint(stage_input_bundle)
        packet_emitted_at = datetime.now(UTC)

        temporal = TemporalContextService().evaluate(stage_input_bundle.temporal)
        regime = RegimeContextService().evaluate(stage_input_bundle.regime)
        options_flow = OptionsFlowContextService().evaluate(stage_input_bundle.options_flow)
        posture = PostureService().evaluate(regime=regime)
        eligibility = PlaybookEligibilityService().evaluate(
            posture=posture, candidate_family_ids=[]
        )
        execution_input = ExecutionExpressionInput(
            temporal=temporal.model_dump(mode="json"),
            regime=regime.model_dump(mode="json"),
            options_flow=options_flow.model_dump(mode="json"),
            posture=posture.model_dump(mode="json"),
            eligibility=eligibility.model_dump(mode="json"),
        )
        execution = evaluate_execution_expression(
            execution_input=execution_input,
            eligibility_output=eligibility,
        )

        stage_outputs = {
            "temporal": temporal,
            "regime": regime,
            "options_flow": options_flow,
            "posture": posture,
            "eligibility": eligibility,
            "execution": execution,
        }
        stage_packets = self._build_stage_packets(
            stage_outputs=stage_outputs,
            source_time_utc=source_time_utc,
            packet_emitted_at=packet_emitted_at,
            traceid=traceid,
            runid=runid,
            scenarioid=scenarioid,
            inputfingerprint=inputfingerprint,
        )

        stage_local_handoff = StageLocalHandoffSurface(
            execution_post_modifier_pre_final_risk=execution.model_dump(mode="json"),
            lineage_tags=list(SERIAL_STAGE_LINEAGE_TAGS),
        )

        return {
            "stage_order": list(self.stage_order),
            "executed_stages": list(self.stage_order),
            "missing_stages": [],
            "serial_spine_contract": copy_serial_spine_contract(),
            "stage_dependencies": {
                stage_name: list(self._stage_dependencies[stage_name])
                for stage_name in self.stage_order
            },
            "stage_packet_ids": {
                stage_name: packet.packet_id for stage_name, packet in stage_packets.items()
            },
            "stage_outputs": {
                stage_name: output.model_dump(mode="json")
                for stage_name, output in stage_outputs.items()
            },
            "stage_packets": {
                stage_name: packet.model_dump(mode="json")
                for stage_name, packet in stage_packets.items()
            },
            "stage_local_handoff": stage_local_handoff.model_dump(mode="json"),
        }

    def _build_stage_packets(
        self,
        *,
        stage_outputs: dict[str, Any],
        source_time_utc: datetime,
        packet_emitted_at: datetime,
        traceid: str,
        runid: str,
        scenarioid: str,
        inputfingerprint: str,
    ) -> dict[str, Any]:
        stage_packet_ids = {
            stage_name: f"dmp::serial::{stage_name}::{_time_token(source_time_utc)}"
            for stage_name in self.stage_order
        }
        packets: dict[str, Any] = {}
        previous_packet_id: str | None = None
        for stage_name in self.stage_order:
            metadata = self._stage_packet_metadata[stage_name]
            dependencyids = [
                stage_packet_ids[name] for name in self._stage_dependencies[stage_name]
            ]
            packet = build_stage_packet(
                packet_id=stage_packet_ids[stage_name],
                dataschema=metadata["dataschema"],
                data=stage_outputs[stage_name],
                emitted_at=packet_emitted_at,
                traceid=traceid,
                runid=runid,
                scenarioid=scenarioid,
                module_id=metadata["module_id"],
                module_version="native",
                module_instance_id=f"{metadata['module_id']}::runtime",
                grammar_role=metadata["grammar_role"],
                stage_name=stage_name,
                parentids=[previous_packet_id] if previous_packet_id is not None else [],
                dependencyids=dependencyids,
                derivedfrom=_derived_from(stage_name, self._stage_dependencies[stage_name]),
                sourceartifactids=[],
                inputfingerprint=inputfingerprint,
                summary_headline=_stage_summary(stage_name, stage_outputs[stage_name]),
                dmpextensions={
                    "nvda": {
                        "schema_identifiers": {
                            "output_model_name": metadata["dataschema"],
                        },
                    }
                },
            )
            packets[stage_name] = packet
            previous_packet_id = packet.packet_id
        return packets


def _attach_runtime_live_inventory(
    receipt: dict[str, Any],
    *,
    live_inventory_service: Any | None,
    live_inventory_request: Mapping[str, Any] | None,
    live_inventory_snapshot: DownstreamLiveInventorySnapshot | Mapping[str, Any] | None,
) -> dict[str, Any]:
    if live_inventory_service is not None and live_inventory_snapshot is not None:
        raise ValueError("live_inventory_service and live_inventory_snapshot cannot both be supplied")
    if live_inventory_service is None and live_inventory_snapshot is None:
        if live_inventory_request is not None:
            raise ValueError("live_inventory_request requires live_inventory_service")
        return receipt
    if live_inventory_snapshot is not None:
        snapshot = _coerce_live_inventory_snapshot(live_inventory_snapshot)
    else:
        request = dict(live_inventory_request or {})
        allowed_keys = {"live_ids", "inventory_id", "capital_snapshot_id", "receipt_id"}
        extra_keys = set(request) - allowed_keys
        if extra_keys:
            joined = ", ".join(sorted(extra_keys))
            raise ValueError(f"live_inventory_request contains unsupported keys: {joined}")
        snapshot = live_inventory_service.build_live_snapshot(**request)
    receipt["runtime_live_inventory"] = snapshot.model_dump(mode="json")
    receipt["runtime_live_inventory_policy_effect"] = (
        "read_only_no_serial_decision_change"
    )
    return receipt


def _coerce_live_inventory_snapshot(
    live_inventory_snapshot: DownstreamLiveInventorySnapshot | Mapping[str, Any],
) -> DownstreamLiveInventorySnapshot:
    if isinstance(live_inventory_snapshot, DownstreamLiveInventorySnapshot):
        return live_inventory_snapshot
    return DownstreamLiveInventorySnapshot.model_validate(live_inventory_snapshot)


def _coerce_admitted_bundle(
    stage_inputs: Mapping[str, Any] | StageInputBundle,
) -> StageInputBundle | None:
    if isinstance(stage_inputs, StageInputBundle):
        return stage_inputs
    if isinstance(stage_inputs, Mapping):
        try:
            return StageInputBundle.model_validate(stage_inputs)
        except ValidationError:
            return None
    return None


def _normalise_stage_inputs(
    stage_inputs: Mapping[str, Any] | StageInputBundle,
) -> Mapping[str, Any]:
    if isinstance(stage_inputs, StageInputBundle):
        return stage_inputs.as_mapping()
    return stage_inputs


def _require_source_time_utc(stage_input_bundle: StageInputBundle) -> datetime:
    source_time_utc = stage_input_bundle.temporal.source_time_utc
    if source_time_utc is None:
        raise ValueError("Serial runtime requires temporal.source_time_utc.")
    if source_time_utc.tzinfo is None:
        return source_time_utc
    return source_time_utc.astimezone(UTC)


def _time_token(source_time_utc: datetime) -> str:
    return source_time_utc.isoformat().replace("+00:00", "Z").replace(":", "").replace("-", "")


def _input_fingerprint(stage_input_bundle: StageInputBundle) -> str:
    payload = json.dumps(
        stage_input_bundle.model_dump(mode="json"), sort_keys=True, separators=(",", ":")
    )
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()


def _derived_from(stage_name: str, dependencies: tuple[str, ...]) -> str:
    if not dependencies:
        return f"{stage_name}_from_declared_stage_inputs"
    dependency_label = "+".join(dependencies)
    return f"{stage_name}_from_{dependency_label}"


def _stage_summary(stage_name: str, payload: Any) -> str:
    if stage_name == "temporal":
        return f"session_state={payload.session_state}"
    if stage_name == "regime":
        return f"volatility_regime={payload.volatility_regime}"
    if stage_name == "options_flow":
        return f"term_structure_state={payload.term_structure_state}"
    if stage_name == "posture":
        return f"permission_state={payload.permission_state.value}"
    if stage_name == "eligibility":
        return f"active_families={','.join(payload.active_family_ids) if payload.active_family_ids else 'none'}"
    if stage_name == "execution":
        return f"lead_playbook_id={payload.lead_playbook_id or 'none'}"
    raise ValueError(f"Unknown stage name: {stage_name}")

from __future__ import annotations

import importlib

import pytest


def _require_module(module_path: str):
    try:
        return importlib.import_module(module_path)
    except ModuleNotFoundError as exc:
        pytest.fail(f"Required module is not implemented: {module_path} ({exc})")


def _require_attr(module, attr_name: str):
    if not hasattr(module, attr_name):
        pytest.fail(f"Required attribute is not implemented: {module.__name__}.{attr_name}")
    return getattr(module, attr_name)


def test_downstream_state_reader_service_surface_is_present() -> None:
    state_reader = _require_module("nvda_desk.downstream.state_reader")
    assert state_reader.__name__ == "nvda_desk.downstream.state_reader"
    assert _require_attr(state_reader, "DownstreamStateService") is getattr(state_reader, "DownstreamStateService")


def test_downstream_state_reader_detaches_position_id_inputs() -> None:
    state_reader = _require_module("nvda_desk.downstream.state_reader")
    assert state_reader.__name__ == "nvda_desk.downstream.state_reader"
    service_class = _require_attr(state_reader, "DownstreamStateService")
    assert service_class is getattr(state_reader, "DownstreamStateService")

    source_position_ids = ["pos-1"]
    state_reader_method_name = "_".join(["read", "state"])
    read_method = getattr(service_class(), state_reader_method_name)
    result = read_method(position_ids=source_position_ids, receipt_id="receipt-1")
    source_position_ids.append("pos-2")
    result.snapshot.position_ids.append("caller-mutated")

    replay = read_method(position_ids=["pos-1"], receipt_id="receipt-1")
    assert replay.snapshot.position_ids == ["pos-1"]
    assert result.receipt_id == "receipt-1"

def _require_module(module_path: str):
    try:
        return importlib.import_module(module_path)
    except ModuleNotFoundError as exc:
        pytest.fail(f"Required module is not implemented: {module_path} ({exc})")


def _require_attr(module, attr_name: str):
    if not hasattr(module, attr_name):
        pytest.fail(f"Required attribute is not implemented: {module.__name__}.{attr_name}")
    return getattr(module, attr_name)
