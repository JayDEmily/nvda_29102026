"""Build deterministic explanation surfaces for downstream review consumers."""

from __future__ import annotations

from nvda_desk.contracts.review import (
    EffectivePolicySnapshot,
    PacketLineageSurface,
    ReviewLineagePacket,
    ReviewPacket,
)


class ReviewExplanationService:
    """Translate review-lineage facts into deterministic explanation text."""

    def build_explanation(
        self,
        *,
        review_packet: ReviewPacket,
        effective_policy: EffectivePolicySnapshot,
        review_lineage: ReviewLineagePacket,
        packet_lineage: PacketLineageSurface,
    ) -> dict:
        summary = review_packet.summary or ""
        resolved_surfaces = list(
            review_lineage.resolved_surfaces or effective_policy.resolved_surfaces
        )
        packet_ids = list(packet_lineage.packet_ids)
        source_packet_ids = list(packet_lineage.source_packet_ids)
        missing_material = []
        if not summary:
            missing_material.append("review_packet.summary")
        if not resolved_surfaces:
            missing_material.append("resolved_surfaces")
        if not packet_ids:
            missing_material.append("packet_lineage.packet_ids")

        surfaces_text = ", ".join(resolved_surfaces)
        packet_ids_text = ", ".join(packet_ids)
        resolved_coefficients = [
            coefficient.model_dump()
            for coefficient in (
                review_lineage.resolved_coefficients or effective_policy.resolved_coefficients
            )
        ]
        return {
            "summary": summary,
            "resolved_surfaces": resolved_surfaces,
            "packet_ids": packet_ids,
            "source_packet_ids": source_packet_ids,
            "resolved_coefficients": resolved_coefficients,
            "missing_review_material": missing_material,
            "lineage_complete": not missing_material,
            "explanation": f"{summary} | surfaces={surfaces_text} | packets={packet_ids_text}".strip(),
        }

from __future__ import annotations

from copy import deepcopy
from typing import Any

from pydantic import BaseModel, ConfigDict, Field, model_validator


class ResolvedCoefficientValue(BaseModel):
    """One resolved coefficient snapshot with baseline/effective lineage."""

    model_config = ConfigDict(extra="forbid")
    coefficient_id: str
    owner_stage: str
    surface_class: str
    baseline_value: float | bool
    effective_value: float | bool
    transform_family: str
    modifier_inputs: list[str] = Field(default_factory=list)
    workbook_ref: str
    runtime_mutable: bool = False

    @model_validator(mode="after")
    def detach_modifier_inputs(self) -> "ResolvedCoefficientValue":
        """Detach caller-owned modifier lineage lists on model construction."""

        self.modifier_inputs = list(self.modifier_inputs)
        return self


def _detached_coefficient(
    coefficient: ResolvedCoefficientValue | dict[str, Any],
) -> ResolvedCoefficientValue:
    if isinstance(coefficient, ResolvedCoefficientValue):
        return ResolvedCoefficientValue.model_validate(deepcopy(coefficient.model_dump()))
    return ResolvedCoefficientValue.model_validate(deepcopy(dict(coefficient)))


class ModifierRuntimePacket(BaseModel):
    model_config = ConfigDict(extra="forbid")
    packet_id: str | None = None
    active_modifier_ids: list[str] = Field(default_factory=list)
    policy_ids: list[str] = Field(default_factory=list)
    resolved_surfaces: list[str] = Field(default_factory=list)
    resolved_coefficients: list[ResolvedCoefficientValue] = Field(default_factory=list)

    @model_validator(mode="after")
    def detach_lineage_lists(self) -> "ModifierRuntimePacket":
        """Detach caller-owned modifier packet lineage containers."""

        self.active_modifier_ids = list(self.active_modifier_ids)
        self.policy_ids = list(self.policy_ids)
        self.resolved_surfaces = list(self.resolved_surfaces)
        self.resolved_coefficients = [
            _detached_coefficient(coefficient)
            for coefficient in self.resolved_coefficients
        ]
        return self


class ModifierReplayReceipt(BaseModel):
    """Deterministic replay receipt for bounded coefficient/modifier authority."""

    model_config = ConfigDict(extra="forbid")
    authority_id: str
    schema_version: str
    owner_stage: str
    active_modifier_ids: list[str] = Field(default_factory=list)
    resolved_coefficients: list[ResolvedCoefficientValue] = Field(default_factory=list)
    replay_fingerprint: str
    closed_authority_classes: list[str] = Field(default_factory=list)

    @model_validator(mode="after")
    def detach_lineage_lists(self) -> "ModifierReplayReceipt":
        """Detach caller-owned replay lineage containers."""

        self.active_modifier_ids = list(self.active_modifier_ids)
        self.resolved_coefficients = [
            _detached_coefficient(coefficient)
            for coefficient in self.resolved_coefficients
        ]
        self.closed_authority_classes = list(self.closed_authority_classes)
        return self

import importlib

def _require_module(module_path: str):
    try:
        return importlib.import_module(module_path)
    except ModuleNotFoundError as exc:
        pytest.fail(f"Required module is not implemented: {module_path} ({exc})")


def _require_attr(module, attr_name: str):
    if not hasattr(module, attr_name):
        pytest.fail(f"Required attribute is not implemented: {module.__name__}.{attr_name}")
    return getattr(module, attr_name)


def test_review_explanation_service_surface_is_present() -> None:
    service = _require_module("nvda_desk.services.review_explanation_service")
    assert service.__name__ == "nvda_desk.services.review_explanation_service"
    assert _require_attr(service, "ReviewExplanationService") is getattr(service, "ReviewExplanationService")


def test_review_explanation_marks_missing_material_without_inventing_lineage() -> None:
    service = ReviewExplanationService()
    explanation = service.build_explanation(
        review_packet=ReviewPacket(),
        effective_policy=EffectivePolicySnapshot(),
        review_lineage=ReviewLineagePacket(),
        packet_lineage=PacketLineageSurface(),
    )

    packet_id_key = "_".join(["packet", "ids"])
    source_packet_id_key = "_".join(["source", "packet", "ids"])

    assert explanation["lineage_complete"] is False
    assert explanation["missing_review_material"] == [
        "review_packet.summary",
        "resolved_surfaces",
        "packet_lineage." + packet_id_key,
    ]
    assert explanation[packet_id_key] == []
    assert explanation[source_packet_id_key] == []


def test_review_explanation_output_is_detached_from_lineage_inputs() -> None:
    review_packet = ReviewPacket(summary="Review ready")
    effective_policy = EffectivePolicySnapshot(resolved_surfaces=["policy_surface"])
    review_lineage = ReviewLineagePacket(resolved_surfaces=["lineage_surface"])
    packet_id_key = "_".join(["packet", "ids"])
    source_packet_id_key = "_".join(["source", "packet", "ids"])
    packet_lineage = PacketLineageSurface(
        **{packet_id_key: ["packet-1"], source_packet_id_key: ["source-1"]}
    )

    output = ReviewExplanationService().build_explanation(
        review_packet=review_packet,
        effective_policy=effective_policy,
        review_lineage=review_lineage,
        packet_lineage=packet_lineage,
    )
    output["resolved_surfaces"].append("mutated")
    output[packet_id_key].append("mutated")

    assert review_lineage.resolved_surfaces == ["lineage_surface"]
    assert getattr(packet_lineage, packet_id_key) == ["packet-1"]
    assert output["lineage_complete"] is True

    from nvda_desk.contracts.modifiers import ResolvedCoefficientValue

        coefficient_id="coef-1",

        surface_class="review_explanation",

        effective_value=1.25,
        transform_family="bounded_review",
        modifier_inputs=["policy-a"],
        workbook_ref="workbook#coef-1",

    output = ReviewExplanationService().build_explanation(
        review_packet=ReviewPacket(summary="Review ready"),

            resolved_surfaces=["lineage_surface"],
            resolved_coefficients=[coefficient],

        packet_lineage=PacketLineageSurface(packet_ids=["packet-1"]),

    output["resolved_coefficients"][0]["effective_value"] = 99.0
    output["resolved_coefficients"][0]["modifier_inputs"].append("mutated")

    assert coefficient.effective_value == 1.25
    assert coefficient.modifier_inputs == ["policy-a"]
    assert output["resolved_coefficients"][0]["coefficient_id"] == "coef-1"
    assert all(isinstance(item, dict) for item in output["resolved_coefficients"])

from __future__ import annotations

from typing import Iterable

from nvda_desk.contracts.modifiers import ResolvedCoefficientValue
from nvda_desk.contracts.review import ReviewLineage, ReviewPacket


class ReviewStageService:
    """Construct review packets and attach deterministic lineage surfaces."""

    def build_review_packet(
        self,
        *,
        packet_id: str | None = None,
        summary: str | None = None,
        sections: Iterable[str] = (),
    ) -> ReviewPacket:
        return build_review_packet(packet_id=packet_id, summary=summary, sections=sections)

    def attach_lineage(
        self,
        *,
        packet: ReviewPacket,
        event_ids: Iterable[str] = (),
        resolved_surfaces: Iterable[str] = (),
        resolved_coefficients: Iterable[ResolvedCoefficientValue] = (),
    ) -> dict:
        return attach_lineage(
            packet=packet,
            event_ids=event_ids,
            resolved_surfaces=resolved_surfaces,
            resolved_coefficients=resolved_coefficients,
        )

    def attach_imported_citations(
        self, *, packet: ReviewPacket, citations: Iterable[str] = ()
    ) -> dict:
        return attach_imported_citations(packet=packet, citations=citations)


def build_review_packet(
    *, packet_id: str | None = None, summary: str | None = None, sections: Iterable[str] = ()
) -> ReviewPacket:
    """Return one typed review packet from runtime-visible text sections."""

    return ReviewPacket(packet_id=packet_id, summary=summary, sections=list(sections))


def attach_lineage(
    *,
    packet: ReviewPacket,
    event_ids: Iterable[str] = (),
    resolved_surfaces: Iterable[str] = (),
    resolved_coefficients: Iterable[ResolvedCoefficientValue] = (),
) -> dict:
    """Attach deterministic lineage metadata to one review packet."""

    lineage = ReviewLineage(
        event_ids=list(event_ids),
        resolved_surfaces=list(resolved_surfaces),
        resolved_coefficients=list(resolved_coefficients),
    )
    return {"review_packet": packet.model_dump(), "lineage": lineage.model_dump()}


def attach_imported_citations(*, packet: ReviewPacket, citations: Iterable[str] = ()) -> dict:
    """Attach import-stage citations without changing review semantics."""

    return {"review_packet": packet.model_dump(), "imported_citations": list(citations)}

"""Replay and comparison services for the bounded analysis-only subsystem."""

from __future__ import annotations

from copy import deepcopy
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Mapping

from nvda_desk.contracts.replay import ReplayPacketLineage

DEFAULT_REPLAY_FIXTURE_ROOT = Path("fixtures/replay")


@dataclass(frozen=True)
class ReplayFixtureSource:
    """Describe the admitted replay fixture root without invoking live runtime flows."""

    fixture_root: Path
    readme_path: Path


@dataclass(frozen=True)
class ReplayComparisonResult:
    """Hold deterministic comparison output for bounded replay/evaluation work."""

    baseline_name: str
    candidate_name: str
    differing_fields: tuple[str, ...]
    baseline_only_fields: tuple[str, ...]
    candidate_only_fields: tuple[str, ...]
    baseline_snapshot: dict[str, Any]
    candidate_snapshot: dict[str, Any]


class ReplayEvaluationService:
    """Build replay lineage and compare prepared or fixture-backed artefacts."""

    def describe_fixture_source(
        self, fixture_root: Path = DEFAULT_REPLAY_FIXTURE_ROOT
    ) -> ReplayFixtureSource:
        resolved_root = fixture_root.resolve()
        return ReplayFixtureSource(
            fixture_root=resolved_root,
            readme_path=(resolved_root / "README.md").resolve(),
        )

    def build_lineage(
        self,
        *,
        packet_ids: list[str],
        replay_run_id: str | None = None,
    ) -> ReplayPacketLineage:
        return ReplayPacketLineage(packet_ids=list(packet_ids), replay_run_id=replay_run_id)

    def compare_outputs(
        self,
        *,
        baseline: Mapping[str, Any],
        candidate: Mapping[str, Any],
        baseline_name: str = "baseline",
        candidate_name: str = "candidate",
    ) -> ReplayComparisonResult:
        baseline_snapshot = deepcopy(dict(baseline))
        candidate_snapshot = deepcopy(dict(candidate))
        baseline_keys = set(baseline_snapshot)
        candidate_keys = set(candidate_snapshot)
        shared_keys = baseline_keys & candidate_keys
        differing_fields = tuple(
            sorted(key for key in shared_keys if baseline_snapshot[key] != candidate_snapshot[key])
        )
        return ReplayComparisonResult(
            baseline_name=baseline_name,
            candidate_name=candidate_name,
            differing_fields=differing_fields,
            baseline_only_fields=tuple(sorted(baseline_keys - candidate_keys)),
            candidate_only_fields=tuple(sorted(candidate_keys - baseline_keys)),
            baseline_snapshot=baseline_snapshot,
            candidate_snapshot=candidate_snapshot,
        )


def describe_replay_fixture_source(
    fixture_root: Path = DEFAULT_REPLAY_FIXTURE_ROOT,
) -> ReplayFixtureSource:
    """Describe the admitted replay fixture root via the bounded analysis service."""

    return ReplayEvaluationService().describe_fixture_source(fixture_root)

"""Persist review packets and lineage records in one bounded in-memory store."""

from __future__ import annotations

from copy import deepcopy
from typing import Any, Mapping


class ReviewRepository:
    """Deterministic in-memory repository for review outputs."""

    def __init__(self):
        self._packets: dict[str, dict[str, Any]] = {}

    def write_packet(self, packet_id: str, packet: Mapping[str, Any]) -> None:
        """Store one review packet by identifier."""

        normalised_id = str(packet_id).strip()
        if not normalised_id:
            raise ValueError("packet_id must be non-empty")
        if normalised_id in self._packets:
            raise ValueError(f"duplicate packet_id: {normalised_id}")
        self._packets[normalised_id] = deepcopy(dict(packet))

    def read_packet(self, packet_id: str) -> dict[str, Any] | None:
        """Return one stored review packet, if present."""

        packet = self._packets.get(str(packet_id).strip())
        return deepcopy(packet) if packet is not None else None

    def list_packets(self) -> list[dict[str, Any]]:
        """Return stored review packets in insertion order."""

        return deepcopy(list(self._packets.values()))

def _imports(path: Path) -> list[str]:
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    imported: list[str] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            imported.extend(alias.name for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            imported.append(node.module)
    return imported


def test_native_review_transform_boundary_excludes_legacy_donor_imports() -> None:
    offenders: list[str] = []
    for path in ACTIVE_REVIEW_FILES:
        for imported in _imports(path):
            if any(imported == prefix or imported.startswith(f"{prefix}.") for prefix in DONOR_IMPORT_PREFIXES):
                offenders.append(f"{path}:{imported}")

    assert offenders == []


def test_native_review_stage_keeps_imported_citations_review_only() -> None:

from __future__ import annotations

from typing import Literal

from pydantic import BaseModel, ConfigDict, Field

CitationAdmissionClass = Literal["admitted_partial", "rejected", "superseded", "reserved"]
CitationDecisionStatus = Literal["accepted_for_review_only", "rejected", "superseded", "reserved"]


class ImportedModuleCitation(BaseModel):
    """One imported-module claim cited during review-only replay work."""

    model_config = ConfigDict(extra="forbid")

    source_identifier: str = Field(min_length=1)
    imported_module: str = Field(min_length=1)
    cited_claim: str = Field(min_length=1)


class ReviewReplayDecision(BaseModel):

    model_config = ConfigDict(extra="forbid")

    source_identifier: str = Field(min_length=1)
    decision_id: str = Field(min_length=1)
    admission_class: CitationAdmissionClass
    status: CitationDecisionStatus
    rejection_reason: str | None = None

    model_config = ConfigDict(extra="forbid")

    source_identifier: str = Field(min_length=1)
    imported_module: str = Field(min_length=1)
    cited_claim: str = Field(min_length=1)
    decision_id: str = Field(min_length=1)
    admission_class: CitationAdmissionClass
    rejection_or_supersession_status: CitationDecisionStatus
    rejection_reason: str | None = None


class ReviewReplayComparisonResult(BaseModel):

    model_config = ConfigDict(extra="forbid")

"""Deterministic review-replay comparison services."""

from __future__ import annotations

from collections.abc import Iterable

from nvda_desk.contracts.review_replay import (
    ImportedModuleCitation,
    ReviewReplayComparisonResult,
    ReviewReplayDecision,
)
from nvda_desk.review.imported_module_citations import map_imported_module_citations


def compare_imported_module_review_replay(
    *,
    citations: Iterable[ImportedModuleCitation],
    decisions: Iterable[ReviewReplayDecision],
) -> ReviewReplayComparisonResult:
    """Compare imported-module citation claims against review decisions without runtime promotion."""

    citation_list = sorted(list(citations), key=lambda row: row.source_identifier)
    decision_sources = {decision.source_identifier for decision in decisions}
    compared_sources = [citation.source_identifier for citation in citation_list]
    missing_sources = sorted(set(compared_sources) - decision_sources)
    traces = []
    if not missing_sources:
        traces = map_imported_module_citations(citations=citation_list, decisions=decisions)
    return ReviewReplayComparisonResult(
        compared_source_identifiers=compared_sources,
        missing_decision_source_identifiers=missing_sources,
        citation_traces=traces,
        promoted_to_runtime=False,
        runtime_side_effects=[],
    )

def _python_files(*roots: str) -> list[Path]:
    files: list[Path] = []
    for root in roots:
        files.extend(sorted(Path(root).rglob("*.py")))
    return files


def _imports_package(path: Path, package_name: str) -> bool:
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                if alias.name == package_name or alias.name.startswith(f"{package_name}."):
                    return True
        if isinstance(node, ast.ImportFrom) and node.module:
            if node.module == package_name or node.module.startswith(f"{package_name}."):
                return True
    return False

from __future__ import annotations

from nvda_desk.contracts.review_replay import ImportedModuleCitation, ReviewReplayDecision
from nvda_desk.replay.review_compare import compare_imported_module_review_replay
from nvda_desk.review.imported_module_citations import map_imported_module_citations


def _citations() -> list[ImportedModuleCitation]:
    return [
        ImportedModuleCitation(
            source_identifier="atlas/module-a.py:10-20",
            imported_module="legacy.module_a",
            cited_claim="legacy module A produced review signal",
        ),
        ImportedModuleCitation(
            source_identifier="atlas/module-b.py:5-7",
            imported_module="legacy.module_b",
            cited_claim="legacy module B was rejected",
        ),
    ]


def test_review_replay_compares_imported_module_claims_without_runtime_promotion() -> None:
    decisions = [
        ReviewReplayDecision(
            source_identifier="atlas/module-a.py:10-20",
            decision_id="decision-a",
            admission_class="admitted_partial",
            status="accepted_for_review_only",
        ),
        ReviewReplayDecision(
            source_identifier="atlas/module-b.py:5-7",
            decision_id="decision-b",
            admission_class="rejected",
            status="superseded",
        ),
    ]

    result = compare_imported_module_review_replay(citations=_citations(), decisions=decisions)

    assert result.promoted_to_runtime is False
    assert result.runtime_side_effects == []
    assert result.compared_source_identifiers == [
        "atlas/module-a.py:10-20",
        "atlas/module-b.py:5-7",
    ]
    assert result.missing_decision_source_identifiers == []


def test_imported_module_citations_are_traceable_to_source_and_decision() -> None:
    mapped = map_imported_module_citations(
        citations=_citations(),
        decisions=[
            ReviewReplayDecision(
                source_identifier="atlas/module-a.py:10-20",
                decision_id="decision-a",
                admission_class="admitted_partial",
                status="accepted_for_review_only",
            ),
            ReviewReplayDecision(
                source_identifier="atlas/module-b.py:5-7",
                decision_id="decision-b",
                admission_class="rejected",
                status="superseded",
                rejection_reason="native owner already exists",
            ),
        ],
    )

    assert [row.source_identifier for row in mapped] == [
        "atlas/module-a.py:10-20",
        "atlas/module-b.py:5-7",
    ]
    assert [row.decision_id for row in mapped] == ["decision-a", "decision-b"]
    assert [row.admission_class for row in mapped] == ["admitted_partial", "rejected"]
    assert mapped[1].rejection_or_supersession_status == "superseded"
    assert mapped[1].rejection_reason == "native owner already exists"


def test_review_replay_result_is_deterministic_for_same_inputs() -> None:
    decisions = [
        ReviewReplayDecision(
            source_identifier="atlas/module-b.py:5-7",
            decision_id="decision-b",
            admission_class="rejected",
            status="superseded",
        ),
        ReviewReplayDecision(
            source_identifier="atlas/module-a.py:10-20",
            decision_id="decision-a",
            admission_class="admitted_partial",
            status="accepted_for_review_only",
        ),
    ]

    first = compare_imported_module_review_replay(citations=_citations(), decisions=decisions)
    second = compare_imported_module_review_replay(
        citations=_citations(), decisions=list(reversed(decisions))
    )

    assert first.model_dump() == second.model_dump()
