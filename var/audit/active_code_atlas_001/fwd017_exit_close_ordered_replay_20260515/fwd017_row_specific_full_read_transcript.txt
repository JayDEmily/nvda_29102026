def find_forbidden_exit_close_authority_keys(value: Any) -> list[str]:
    """Return forbidden authority-key names found anywhere in a nested payload."""

    found: set[str] = set()

    def walk(node: Any) -> None:
        if isinstance(node, dict):
            for key, child in node.items():
                key_text = str(key)
                if key_text in EXIT_CLOSE_FORBIDDEN_AUTHORITY_KEYS:
                    found.add(key_text)
                walk(child)
        elif isinstance(node, (list, tuple, set)):
            for child in node:
                walk(child)

    walk(value)
    return sorted(found)


def _fingerprint(payload: dict[str, Any]) -> str:
    canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"), default=str)
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()

class ExitClosePolicyDecision(BaseModel):
    """Bounded exit-policy decision without broker execution."""

    model_config = ConfigDict(extra="forbid")

    decision_id: str = Field(min_length=1)
    position_id: str = Field(min_length=1)
    requested_action: ExitCloseAction
    exit_state: ExitCloseState
    trigger_ids: tuple[str, ...] = Field(default_factory=tuple)
    blocked_reasons: tuple[str, ...] = Field(default_factory=tuple)
    input_trace: ExitCloseInputTrace
    broker_execution_opened: bool = False
    live_inventory_mutated: bool = False
    capital_sizing_recomputed: bool = False

    @model_validator(mode="after")
    def _validate_state_consistency(self) -> "ExitClosePolicyDecision":
        expected_state = {
            "hold": "hold",
            "trim": "trim_ready",
            "close": "close_ready",
        }[self.requested_action]
        if self.blocked_reasons:
            if self.exit_state != "blocked":
                raise ValueError("blocked exit/close decisions must use exit_state='blocked'")
        elif self.exit_state != expected_state:
            raise ValueError("exit_state must match requested_action when not blocked")
        if self.broker_execution_opened:
            raise ValueError("exit/close policy decision must not open broker execution")
        if self.live_inventory_mutated:
            raise ValueError("exit/close policy decision must not mutate live inventory")
        if self.capital_sizing_recomputed:
            raise ValueError("exit/close policy decision must not recompute capital sizing")
        return self


class CloseHandoffMaterial(BaseModel):
    """Protocol-only close handoff material for a future broker boundary."""

    model_config = ConfigDict(extra="forbid")

    handoff_id: str = Field(min_length=1)
    decision_id: str = Field(min_length=1)
    position_id: str = Field(min_length=1)
    close_intent_preview: dict[str, Any] = Field(default_factory=dict)
    broker_boundary_mode: str = "protocol_only"
    broker_protocol_only: bool = True
    execution_allowed: bool = False
    live_order_placement_opened: bool = False

    @model_validator(mode="after")
    def _validate_protocol_only(self) -> "CloseHandoffMaterial":
        if self.broker_boundary_mode != "protocol_only":
            raise ValueError("close handoff may use broker_boundary_mode='protocol_only' only")
        if (
            not self.broker_protocol_only
            or self.execution_allowed
            or self.live_order_placement_opened
        ):
            raise ValueError("close handoff must remain non-executing protocol-only material")
        forbidden = find_forbidden_exit_close_authority_keys(self.close_intent_preview)
        if forbidden:
            raise ValueError(
                "close handoff preview contains forbidden authority keys: " + ", ".join(forbidden)
            )
        return self


class ExitCloseReceipt(BaseModel):
    """Receipt tying one exit/close decision to replayable position-closure evidence."""

    model_config = ConfigDict(extra="forbid")

    receipt_id: str = Field(min_length=1)
    decision: ExitClosePolicyDecision
    input_trace: ExitCloseInputTrace
    close_handoff: CloseHandoffMaterial | None = None
    emitted_output: dict[str, Any] = Field(default_factory=dict)
    position_closure_replay_fingerprint: str
    broker_execution_opened: bool = False
    live_inventory_mutated: bool = False

    @model_validator(mode="after")
    def _validate_receipt_boundary(self) -> "ExitCloseReceipt":
        if self.broker_execution_opened:
            raise ValueError("exit/close receipt must not open broker execution")
        if self.live_inventory_mutated:
            raise ValueError("exit/close receipt must not mutate live inventory")
        if self.input_trace != self.decision.input_trace:
            raise ValueError("exit/close receipt input_trace must match decision input_trace")
        forbidden = find_forbidden_exit_close_authority_keys(self.emitted_output)
        if forbidden:
            raise ValueError(
                "exit/close emitted output contains forbidden authority keys: "
                + ", ".join(forbidden)
            )
        return self


def fingerprint_exit_close_replay(
    *,
    decision: ExitClosePolicyDecision,
    close_handoff: CloseHandoffMaterial | None = None,
) -> str:
    """Return deterministic replay fingerprint for a bounded exit/close result."""

    return _fingerprint(
        {
            "decision": decision.model_dump(mode="json"),
            "close_handoff": close_handoff.model_dump(mode="json") if close_handoff else None,
        }
    )

class ExitCloseService:
    """Own bounded row 5.14 exit/close evaluation without broker execution."""

    def evaluate_policy(
        self,
        *,
        input_trace: ExitCloseInputTrace,
        requested_action: ExitCloseAction,
        trigger_ids: tuple[str, ...] = (),
        decision_id: str = "exit-close-decision",
        position_open: bool = True,
    ) -> ExitClosePolicyDecision:
        return evaluate_exit_close_policy(
            input_trace=input_trace,
            requested_action=requested_action,
            trigger_ids=trigger_ids,
            decision_id=decision_id,
            position_open=position_open,
        )

    def build_handoff(
        self,
        *,
        decision: ExitClosePolicyDecision,
        broker_boundary_mode: str = "protocol_only",
    ) -> CloseHandoffMaterial:
        return build_close_handoff(
            decision=decision,
            broker_boundary_mode=broker_boundary_mode,
        )

    def build_receipt(
        self,
        *,
        receipt_id: str,
        decision: ExitClosePolicyDecision,
        close_handoff: CloseHandoffMaterial | None = None,
    ) -> ExitCloseReceipt:
        return build_exit_close_receipt(
            receipt_id=receipt_id,
            decision=decision,
            close_handoff=close_handoff,
        )


def evaluate_exit_close_policy(
    *,
    input_trace: ExitCloseInputTrace,
    requested_action: ExitCloseAction,
    trigger_ids: tuple[str, ...] = (),
    decision_id: str = "exit-close-decision",
    position_open: bool = True,
) -> ExitClosePolicyDecision:
    """Return a bounded exit/close policy decision from admitted downstream inputs."""

    blocked_reasons: tuple[str, ...] = ()
    if not position_open:
        blocked_reasons = ("position_not_open",)
    elif requested_action in {"trim", "close"} and not trigger_ids:
        blocked_reasons = ("exit_or_close_trigger_required",)

    exit_state = "blocked"
    if not blocked_reasons:
        exit_state = {
            "hold": "hold",
            "trim": "trim_ready",
            "close": "close_ready",
        }[requested_action]

    return ExitClosePolicyDecision(
        decision_id=decision_id,
        position_id=input_trace.position_id,
        requested_action=requested_action,
        exit_state=exit_state,
        trigger_ids=trigger_ids,
        blocked_reasons=blocked_reasons,
        input_trace=input_trace,
    )


def build_close_handoff(
    *,
    decision: ExitClosePolicyDecision,
    broker_boundary_mode: str = "protocol_only",
) -> CloseHandoffMaterial:
    """Return protocol-only handoff material for trim/close-ready decisions."""

    if decision.exit_state not in {"trim_ready", "close_ready"}:
        raise ValueError("close handoff requires trim_ready or close_ready decision")
    return CloseHandoffMaterial(
        handoff_id=f"{decision.decision_id}::close-handoff",
        decision_id=decision.decision_id,
        position_id=decision.position_id,
        broker_boundary_mode=broker_boundary_mode,
        close_intent_preview={
            "position_id": decision.position_id,
            "decision_id": decision.decision_id,
            "requested_action": decision.requested_action,
            "exit_state": decision.exit_state,
            "trigger_ids": list(decision.trigger_ids),
            "close_policy_id": decision.input_trace.close_policy_id,
        },
    )


def build_exit_close_receipt(
    *,
    receipt_id: str,
    decision: ExitClosePolicyDecision,
    close_handoff: CloseHandoffMaterial | None = None,
) -> ExitCloseReceipt:
    """Return replayable exit/close receipt without live order placement."""

    fingerprint = fingerprint_exit_close_replay(
        decision=decision,
        close_handoff=close_handoff,
    )
    return ExitCloseReceipt(
        receipt_id=receipt_id,
        decision=decision,
        input_trace=decision.input_trace,
        close_handoff=close_handoff,
        emitted_output={
            "decision_id": decision.decision_id,
            "position_id": decision.position_id,
            "requested_action": decision.requested_action,
            "exit_state": decision.exit_state,
            "blocked_reasons": list(decision.blocked_reasons),
            "close_handoff_id": close_handoff.handoff_id if close_handoff else None,
        },
        position_closure_replay_fingerprint=fingerprint,
    )


def replay_exit_close_receipt(receipt: ExitCloseReceipt) -> str:
    """Recompute the replay fingerprint for a previously emitted exit/close receipt."""

    return fingerprint_exit_close_replay(
        decision=receipt.decision,
        close_handoff=receipt.close_handoff,
    )

def _payload_keys(value: Any):
    if isinstance(value, dict):
        for key, child in value.items():
            yield str(key)
            yield from _payload_keys(child)
    elif isinstance(value, (list, tuple, set)):
        for child in value:
            yield from _payload_keys(child)


def _reject_live_inventory_transform_required_donor_payload(value: Any):
    observed = set(_payload_keys(value))

            "live inventory source_payload cannot carry transform-required donor "

    return value

def _trace(
    *,
    decision: CapitalDeploymentDecision,
    serial_candidate_id: str,
    placement_constraint_ids: list[str],
    broker_boundary_mode: str,
) -> PlacementReadinessInputTrace:
    receipt_id = decision.capital_deployment_receipt_id
    if not receipt_id:
        raise ValueError("placement readiness requires capital_deployment_receipt_id")
    if not serial_candidate_id.strip():
        raise ValueError("placement readiness requires serial_candidate_id")
    return PlacementReadinessInputTrace(
        capital_deployment_receipt_id=receipt_id,
        serial_candidate_id=serial_candidate_id,
        placement_constraint_ids=list(placement_constraint_ids),
        broker_boundary_mode=broker_boundary_mode,
    )


def evaluate_placement_readiness(
    *,
    decision: CapitalDeploymentDecision,
    serial_candidate_id: str,
    placement_constraint_ids: list[str] | None = None,
    broker_boundary_mode: str = "protocol_only",
    readiness_id: str = "placement-readiness",
) -> PlacementReadinessResult:
    """Return readiness/handoff material without opening broker execution."""

    _require_protocol_only_boundary(broker_boundary_mode)
    constraints = list(placement_constraint_ids or [])
    trace = _trace(
        decision=decision,
        serial_candidate_id=serial_candidate_id,
        placement_constraint_ids=constraints,
        broker_boundary_mode=broker_boundary_mode,
    )
    blocked_reasons: list[str] = []
    if decision.action != CapitalDeploymentAuthorityAction.CONTINUE:
        blocked_reasons.append(f"capital_deployment_action_{decision.action.value}")
    if decision.authorised_deployable_pct <= 0.0:
        blocked_reasons.append("no_authorised_deployable_percentage")
    if decision.authorised_notional_usd <= 0.0:
        blocked_reasons.append("no_authorised_notional")

    if blocked_reasons:
        return PlacementReadinessResult(
            readiness_id=readiness_id,
            readiness_state="blocked",
            blocked_reasons=blocked_reasons,
            input_trace=trace,
        )

    handoff = PlacementHandoffMaterial(
        handoff_id=f"{readiness_id}::handoff",
        order_intent_preview={
            "serial_candidate_id": serial_candidate_id,
            "capital_deployment_receipt_id": decision.capital_deployment_receipt_id,
            "authorised_deployable_pct": decision.authorised_deployable_pct,
            "authorised_notional_usd": decision.authorised_notional_usd,
            "placement_constraint_ids": constraints,
        },
    )
    return PlacementReadinessResult(
        readiness_id=readiness_id,
        readiness_state="ready_for_protocol_handoff",
        input_trace=trace,
        placement_handoff=handoff,
    )

"""Build bounded live-inventory snapshots."""

from __future__ import annotations

from collections.abc import Iterable, Mapping
from copy import deepcopy
from typing import Any

from nvda_desk.contracts.downstream import (
    DownstreamLiveInventorySnapshot,
    InventoryExposureState,
    InventoryStateReceipt,
    LiveCapitalState,
    LivePositionState,
)
from nvda_desk.persistence.receipt_ledger import ReceiptLedger
from nvda_desk.repositories.capital_state_repository import CapitalStateRepository
from nvda_desk.repositories.position_repository import PositionRepository


class DownstreamLiveInventoryService:
    """Create truthful live-inventory snapshots without injecting policy."""

    def __init__(
        self,
        *,
        position_repository: PositionRepository | None = None,
        capital_state_repository: CapitalStateRepository | None = None,
        receipt_ledger: ReceiptLedger | None = None,
    ):
        self._position_repository = position_repository
        self._capital_state_repository = capital_state_repository
        self._receipt_ledger = receipt_ledger

    def build_live_snapshot(
        self,
        live_ids: list[str] | None = None,
        *,
        inventory_id: str = "live-inventory",
        capital_snapshot_id: str | None = None,
        receipt_id: str | None = None,
    ) -> DownstreamLiveInventorySnapshot:
        """Build one inventory snapshot from repo-native state without policy decisions."""

        requested_ids = _normalise_id_list(live_ids)
        position_states = self._read_position_states(requested_ids)
        capital_state = self._read_capital_state(capital_snapshot_id)
        exposure_state = _build_exposure_state(position_states)
        receipt = None
        if receipt_id is not None:
            receipt = InventoryStateReceipt(
                receipt_id=_normalise_token(receipt_id, "receipt_id"),
                inventory_id=_normalise_token(inventory_id, "inventory_id"),
                position_ids=[state.position_id for state in position_states],
                capital_snapshot_id=capital_state.snapshot_id if capital_state else None,
            )
            if self._receipt_ledger is not None:
                self._receipt_ledger.append_receipt(
                    receipt.receipt_id,
                    "live_inventory_snapshot",
                    receipt.model_dump(),
                )
        return DownstreamLiveInventorySnapshot(
            inventory_id=_normalise_token(inventory_id, "inventory_id"),
            live_ids=[state.position_id for state in position_states] or requested_ids,
            position_states=position_states,
            capital_state=capital_state,
            exposure_state=exposure_state,
            inventory_receipt=receipt,
        )

    def reconcile_fill_evidence(
        self,
        fill_evidence: Iterable[Mapping[str, Any]],
        *,
        reconciliation_id: str,
        inventory_id: str = "live-inventory",
        capital_snapshot_id: str | None = None,
        receipt_id: str | None = None,
    ) -> DownstreamLiveInventorySnapshot:
        """Reconcile executed fill evidence into repo-native live-inventory state."""

        if self._position_repository is None:
            raise ValueError("position_repository is required for fill reconciliation")

        normalised_reconciliation_id = _normalise_token(
            reconciliation_id, "reconciliation_id"
        )
        normalised_events = [_normalise_fill_evidence_event(event) for event in fill_evidence]
        if not normalised_events:
            raise ValueError("fill_evidence must contain at least one event")
        _assert_unique_fill_evidence_ids(normalised_events)

        grouped_payloads: dict[str, dict[str, Any]] = {}
        for event in normalised_events:
            position_id = event["position_id"]
            payload = grouped_payloads.get(position_id)
            if payload is None:
                existing = self._position_repository.read_position(position_id)
                payload = deepcopy(existing) if existing is not None else {}
                payload.setdefault("fill_evidence_ids", [])
                payload.setdefault("fill_evidence", [])
                grouped_payloads[position_id] = payload

            existing_fill_ids = _normalised_existing_fill_ids(payload.get("fill_evidence_ids"))
            if event["fill_evidence_id"] in existing_fill_ids:
                raise ValueError(f"duplicate fill_evidence_id: {event['fill_evidence_id']}")

            if event.get("symbol") and not payload.get("symbol"):
                payload["symbol"] = event["symbol"]

            declared_delta = _optional_float(event.get("declared_exposure_delta"))
            if declared_delta is not None:
                current_declared = _optional_float(payload.get("declared_exposure")) or 0.0
                payload["declared_exposure"] = current_declared + declared_delta

            payload["fill_evidence_ids"].append(event["fill_evidence_id"])
            payload["fill_evidence"].append(deepcopy(event))
            payload["inventory_reconciliation_id"] = normalised_reconciliation_id

        for position_id, payload in grouped_payloads.items():
            if self._position_repository.read_position(position_id) is None:
                self._position_repository.write_position(position_id, payload)
            else:
                self._position_repository.replace_position(position_id, payload)

        reconciled_position_ids = list(grouped_payloads)
        snapshot = self.build_live_snapshot(
            live_ids=reconciled_position_ids,
            inventory_id=inventory_id,
            capital_snapshot_id=capital_snapshot_id,
        )
        normalised_receipt_id = receipt_id
        receipt = None
        if normalised_receipt_id is not None:
            receipt = InventoryStateReceipt(
                receipt_id=_normalise_token(normalised_receipt_id, "receipt_id"),
                inventory_id=_normalise_token(inventory_id, "inventory_id"),
                position_ids=reconciled_position_ids,
                capital_snapshot_id=snapshot.capital_state.snapshot_id
                if snapshot.capital_state
                else None,
            )
            if self._receipt_ledger is not None:
                receipt_payload = receipt.model_dump()
                receipt_payload.update(
                    {
                        "reconciliation_id": normalised_reconciliation_id,
                        "fill_evidence_ids": [
                            event["fill_evidence_id"] for event in normalised_events
                        ],
                    }
                )
                self._receipt_ledger.append_receipt(
                    receipt.receipt_id,
                    "live_inventory_reconciliation",
                    receipt_payload,
                )
        return snapshot.model_copy(update={"inventory_receipt": receipt})

    def _read_position_states(self, requested_ids: list[str]) -> list[LivePositionState]:
        if self._position_repository is None:
            return [LivePositionState(position_id=position_id) for position_id in requested_ids]

        if requested_ids:
            records = [
                self._position_repository.read_position_record(position_id)
                for position_id in requested_ids
            ]
            records = [record for record in records if record is not None]
        else:
            records = self._position_repository.list_position_records()

        return [
            LivePositionState(
                position_id=str(record["position_id"]),
                source_payload=deepcopy(dict(record["position"])),
            )
            for record in records
        ]

    def _read_capital_state(self, snapshot_id: str | None) -> LiveCapitalState | None:
        if self._capital_state_repository is None:
            return None
        if snapshot_id is None:
            records = self._capital_state_repository.list_snapshot_records()
            if not records:
                return None
            record = records[-1]
        else:
            record = self._capital_state_repository.read_snapshot_record(snapshot_id)
            if record is None:
                return None
        return LiveCapitalState(
            snapshot_id=str(record["snapshot_id"]),
            source_payload=deepcopy(dict(record["snapshot"])),
        )


_FILL_EVIDENCE_FORBIDDEN_KEYS = frozenset(
    {
        "broker_order_id",
        "order_submitted",
        "execution_allowed",
        "capital_deployment_receipt_id",
        "opening_authorisation",
        "placement_handoff",
    }
)


def _normalise_fill_evidence_event(event: Mapping[str, Any]) -> dict[str, Any]:
    payload = deepcopy(dict(event))
    forbidden = set(payload) & _FILL_EVIDENCE_FORBIDDEN_KEYS
    if forbidden:
        joined = ", ".join(sorted(forbidden))
        raise ValueError(f"fill evidence cannot carry placement authority keys: {joined}")
    payload["fill_evidence_id"] = _normalise_token(
        payload.get("fill_evidence_id", ""), "fill_evidence_id"
    )
    payload["position_id"] = _normalise_token(payload.get("position_id", ""), "position_id")
    if "symbol" in payload:
        symbol = _optional_text(payload["symbol"])
        if symbol is not None:
            payload["symbol"] = symbol
        else:
            payload.pop("symbol")
    declared_delta = _optional_float(payload.get("declared_exposure_delta"))
    if declared_delta is not None:
        payload["declared_exposure_delta"] = declared_delta
    return payload


def _assert_unique_fill_evidence_ids(events: list[dict[str, Any]]) -> None:
    observed: set[str] = set()
    for event in events:
        evidence_id = event["fill_evidence_id"]
        if evidence_id in observed:
            raise ValueError(f"duplicate fill_evidence_id: {evidence_id}")
        observed.add(evidence_id)


def _normalised_existing_fill_ids(values: Any) -> set[str]:
    if values is None:
        return set()
    if not isinstance(values, list):
        raise ValueError("fill_evidence_ids must be a list")
    return {_normalise_token(value, "fill_evidence_id") for value in values}


def _build_exposure_state(position_states: Iterable[LivePositionState]) -> InventoryExposureState:
    ids: list[str] = []
    exposures: dict[str, float] = {}
    for state in position_states:
        ids.append(state.position_id)
        symbol = _optional_text(state.source_payload.get("symbol"))
        declared_exposure = _optional_float(state.source_payload.get("declared_exposure"))
        if symbol and declared_exposure is not None:
            exposures[symbol] = exposures.get(symbol, 0.0) + declared_exposure
    return InventoryExposureState(
        position_count=len(ids),
        open_position_ids=ids,
        declared_exposure_by_symbol=exposures,
    )


def _normalise_id_list(values: list[str] | None) -> list[str]:
    if values is None:
        return []
    return [_normalise_token(value, "live_id") for value in values]


def _normalise_token(value: str, field_name: str) -> str:
    token = str(value).strip()
    if not token:
        raise ValueError(f"{field_name} must be non-empty")
    return token


def _optional_text(value: Any) -> str | None:
    if value is None:
        return None
    text = str(value).strip()
    return text or None


def _optional_float(value: Any) -> float | None:
    if value is None:
        return None
    if isinstance(value, bool):
        return None
    try:
        return float(value)
    except (TypeError, ValueError):
        return None

from __future__ import annotations

from enum import StrEnum
from typing import Any

from pydantic import BaseModel, ConfigDict, Field, field_validator, model_validator

CAPITAL_DEPLOYMENT_DECISION_DATASCHEMA = (
    "urn:nvda:schema:capital_deployment_authority.decision:1.0.0"
)

_CAPITAL_RECEIPT_OPENING_OR_BROKER_AUTHORITY_KEYS = frozenset(
    {
        "broker_order_id",
        "execution_allowed",
        "opening_authorisation",
        "order_intent_preview",
        "placement_handoff",
        "readiness_state",
    }
)


def _payload_keys(value: Any):
    if isinstance(value, dict):
        for key, child in value.items():
            yield str(key)
            yield from _payload_keys(child)
    elif isinstance(value, (list, tuple, set)):
        for child in value:
            yield from _payload_keys(child)


def _reject_opening_or_broker_authority_payload(value: Any):
    observed = set(_payload_keys(value))
    forbidden = observed & _CAPITAL_RECEIPT_OPENING_OR_BROKER_AUTHORITY_KEYS
    if forbidden:
        joined = ", ".join(sorted(forbidden))
        raise ValueError(
            "capital deployment receipt emitted_output cannot carry opening or "
            f"broker authority keys: {joined}"
        )
    return value


class CapitalDeploymentAuthorityAction(StrEnum):

    NO_TRADE = "no_trade"
    REDUCE = "reduce"
    CONTINUE = "continue"
    BLOCK = "block"

class CapitalDeploymentAuthorityInput(BaseModel):
    """Inputs consumed by downstream capital deployment authority.

    The model consumes preserved serial-candidate truth, downstream
    pre-deployment truth, parallel-risk envelope truth, final-risk truth,
    explicit capital state, bounded policy, coefficient/modifier bounds, and
    temporal/context signals. Serial confidence is accepted only as auditable
    candidate metadata and is not used as a position-size or notional-sizing
    input.
    """

    model_config = ConfigDict(extra="forbid")

    final_risk_action: str
    final_risk_receipt_id: str
    serial_candidate_id: str
    serial_candidate_eligible: bool = True
    serial_confidence: float | None = Field(default=None, ge=0.0, le=1.0)
    downstream_pre_deployment_allowed: bool = True
    downstream_pre_deployment_receipt_id: str
    parallel_risk_packet_id: str | None = None
    parallel_risk_envelope_state: str = "broad"
    parallel_risk_hard_blocks: list[str] = Field(default_factory=list)
    parallel_risk_caution_flags: list[str] = Field(default_factory=list)
    coefficient_min_deployable_floor_pct: float = Field(default=0.0, ge=0.0, le=100.0)
    temporal_event_window_active: bool = False
    temporal_event_risk: str = "normal"
    capital_state_snapshot_id: str
    available_capital_usd: float = Field(ge=0.0)
    bounded_policy_id: str
    requested_deployable_pct: float = Field(ge=0.0, le=100.0)
    max_deployable_pct: float = Field(ge=0.0, le=100.0)
    inventory_pct: float = Field(default=0.0, ge=0.0, le=100.0)

    @field_validator(
        "final_risk_action",
        "final_risk_receipt_id",
        "serial_candidate_id",
        "downstream_pre_deployment_receipt_id",
        "capital_state_snapshot_id",
        "bounded_policy_id",
    )
    @classmethod
    def validate_required_token(cls, value: str) -> str:
        """Require trace tokens to be explicit and non-empty."""
        token = str(value).strip()
        if not token:
            raise ValueError("capital deployment authority trace tokens must be non-empty")
        if token == "unknown":
            raise ValueError("capital deployment authority trace tokens must not be 'unknown'")
        return token

    @field_validator("parallel_risk_packet_id")
    @classmethod
    def validate_optional_trace_token(cls, value: str | None) -> str | None:
        """Require optional trace tokens to be non-empty when present."""
        if value is None:
            return None
        token = str(value).strip()
        if not token:
            raise ValueError("optional trace tokens must be non-empty when present")
        if token == "unknown":
            raise ValueError("optional trace tokens must not be 'unknown'")
        return token

    @model_validator(mode="after")
    def validate_deployable_floor(self) -> "CapitalDeploymentAuthorityInput":
        """Ensure the coefficient floor does not exceed the risk-authorised cap."""
        if (
            self.coefficient_min_deployable_floor_pct > 0.0
            and self.max_deployable_pct > 0.0
            and self.coefficient_min_deployable_floor_pct > self.max_deployable_pct
        ):
            raise ValueError(
                "coefficient_min_deployable_floor_pct cannot exceed max_deployable_pct"
            )
        return self

"""Bounded downstream capital deployment authority helpers."""

from __future__ import annotations

from nvda_desk.contracts.capital import (
    CapitalDeploymentAuthorityAction,
    CapitalDeploymentAuthorityInput,
    CapitalDeploymentDecision,
    CapitalDeploymentInputTrace,
    CapitalDeploymentReceipt,
)

_CONSUMED_INPUTS = [
    "serial_candidate",
    "downstream_pre_deployment_result",
    "parallel_risk_envelope",
    "final_risk_product",
    "capital_state",
    "bounded_deployment_policy",
    "coefficient_modifier_authority",
    "temporal_context",
]


class CapitalDeploymentAuthorityService:
    """Own bounded downstream capital-deployment authority evaluation."""

    def evaluate(
        self,
        context: CapitalDeploymentAuthorityInput,
        *,
        receipt_id: str | None = None,
    ) -> CapitalDeploymentDecision:
        """Return one bounded downstream capital deployment decision."""
        return evaluate_capital_deployment(context, receipt_id=receipt_id)

    def evaluate_with_receipt(
        self,
        context: CapitalDeploymentAuthorityInput,
        *,
        receipt_id: str,
    ) -> CapitalDeploymentReceipt:
        """Return one receipt tying the decision to its consumed inputs."""
        decision = self.evaluate(context, receipt_id=receipt_id)
        return build_capital_deployment_receipt(
            receipt_id=receipt_id,
            context=context,
            decision=decision,
        )


def _bounded_deployable_pct(context: CapitalDeploymentAuthorityInput) -> float:
    """Return the deployment percentage allowed by downstream risk and policy bounds."""
    return min(context.requested_deployable_pct, context.max_deployable_pct)


def _authorised_notional(context: CapitalDeploymentAuthorityInput, deployable_pct: float) -> float:
    """Return authorised notional from explicit capital state and bounded percentage."""
    return round(context.available_capital_usd * (deployable_pct / 100.0), 2)


def _input_trace(context: CapitalDeploymentAuthorityInput) -> CapitalDeploymentInputTrace:
    """Return the explicit input trace consumed by capital deployment authority."""
    return CapitalDeploymentInputTrace(
        serial_candidate_id=context.serial_candidate_id,
        downstream_pre_deployment_receipt_id=context.downstream_pre_deployment_receipt_id,
        parallel_risk_packet_id=context.parallel_risk_packet_id,
        final_risk_receipt_id=context.final_risk_receipt_id,
        capital_state_snapshot_id=context.capital_state_snapshot_id,
        bounded_policy_id=context.bounded_policy_id,
        consumed_inputs=list(_CONSUMED_INPUTS),
    )


def _decision(
    context: CapitalDeploymentAuthorityInput,
    *,
    action: CapitalDeploymentAuthorityAction,
    deployable_pct: float = 0.0,
    notional_usd: float = 0.0,
    rationale: list[str],
    receipt_id: str | None = None,
) -> CapitalDeploymentDecision:
    """Return an auditable decision carrying the closed Sequence 9 action vocabulary."""
    return CapitalDeploymentDecision(
        action=action,
        authorised_deployable_pct=deployable_pct,
        authorised_notional_usd=notional_usd,
        consumed_inputs=list(_CONSUMED_INPUTS),
        rationale=rationale,
        input_trace=_input_trace(context),
        capital_deployment_receipt_id=receipt_id,
    )


def evaluate_capital_deployment(
    context: CapitalDeploymentAuthorityInput,
    *,
    receipt_id: str | None = None,
) -> CapitalDeploymentDecision:
    """Return one bounded downstream capital deployment decision.

    Precedence is deterministic: downstream pre-deployment blocks dominate,
    parallel/final risk hard blocks dominate, serial ineligibility produces
    no-trade, risk or coefficient compression reduces deployment, temporal
    elevated event risk reduces deployment, and only a clean bounded context
    continues. Serial confidence is deliberately not used for sizing.
    """
    if not context.downstream_pre_deployment_allowed:
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.BLOCK,
            rationale=["downstream_pre_deployment_blocked_candidate"],
            receipt_id=receipt_id,
        )

    if (
        context.final_risk_action != "proceed"
        or context.parallel_risk_envelope_state == "block"
        or bool(context.parallel_risk_hard_blocks)
    ):
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.BLOCK,
            rationale=["parallel_or_final_risk_block_dominates"],
            receipt_id=receipt_id,
        )

    if not context.serial_candidate_eligible:
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.NO_TRADE,
            rationale=["serial_candidate_not_eligible"],
            receipt_id=receipt_id,
        )

    deployable_pct = _bounded_deployable_pct(context)
    authorised = _authorised_notional(context, deployable_pct)

    if deployable_pct <= 0.0 or authorised <= 0.0:
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.NO_TRADE,
            rationale=["no_positive_downstream_deployment_authorised"],
            receipt_id=receipt_id,
        )

    if deployable_pct < context.requested_deployable_pct:
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.REDUCE,
            deployable_pct=deployable_pct,
            notional_usd=authorised,
            rationale=["parallel_risk_envelope_reduced_requested_deployment"],
            receipt_id=receipt_id,
        )

    if deployable_pct < context.coefficient_min_deployable_floor_pct:
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.REDUCE,
            deployable_pct=deployable_pct,
            notional_usd=authorised,
            rationale=["coefficient_floor_not_satisfied"],
            receipt_id=receipt_id,
        )

    if context.parallel_risk_caution_flags or context.parallel_risk_envelope_state in {
        "tight",
        "caution",
    }:
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.REDUCE,
            deployable_pct=deployable_pct,
            notional_usd=authorised,
            rationale=["parallel_risk_caution_requires_reduction"],
            receipt_id=receipt_id,
        )

    if context.temporal_event_window_active and context.temporal_event_risk == "elevated":
        return _decision(
            context,
            action=CapitalDeploymentAuthorityAction.REDUCE,
            deployable_pct=deployable_pct,
            notional_usd=authorised,
            rationale=["temporal_event_window_elevated_risk"],
            receipt_id=receipt_id,
        )

    return _decision(
        context,
        action=CapitalDeploymentAuthorityAction.CONTINUE,
        deployable_pct=deployable_pct,
        notional_usd=authorised,
        rationale=["serial_parallel_coefficient_and_temporal_inputs_allow_continuation"],
        receipt_id=receipt_id,
    )


def build_capital_deployment_receipt(
    *,

    )

def _require_module(module_path: str):
    try:
        return importlib.import_module(module_path)
    except ModuleNotFoundError as exc:
        pytest.fail(f"Required module is not implemented: {module_path} ({exc})")


def _require_attr(module, attr_name: str):
    if not hasattr(module, attr_name):
        pytest.fail(f"Required attribute is not implemented: {module.__name__}.{attr_name}")
    return getattr(module, attr_name)

    try:
        return importlib.import_module(module_path)
    except ModuleNotFoundError as exc:
        pytest.fail(f"Required module is not implemented: {module_path} ({exc})")

    if not hasattr(module, attr_name):
