    18	def _require_module(module_path: str):
    19	    try:
    20	        return importlib.import_module(module_path)
    21	    except ModuleNotFoundError as exc:
    22	        pytest.fail(f"Required module is not implemented: {module_path} ({exc})")
    23	
    24	
    25	def _require_attr(module, attr_name: str):
    26	    if not hasattr(module, attr_name):
    27	        pytest.fail(f"Required attribute is not implemented: {module.__name__}.{attr_name}")
    28	    return getattr(module, attr_name)
    29

    37	def test_live_inventory_snapshot_reads_position_and_capital_state_copy_safely() -> None:
    38	    positions = PositionRepository()
    39	    capital = CapitalStateRepository()
    40	    positions.write_position(
    41	        "pos-1",
    42	        {"symbol": "NVDA", "declared_exposure": 12.5, "metadata": {"source": "fill"}},
    43	    )
    44	    capital.write_snapshot("cap-1", {"cash": 1000.0, "metadata": {"source": "ledger"}})
    45	
    46	    snapshot = DownstreamLiveInventoryService(
    47	        position_repository=positions,
    48	        capital_state_repository=capital,
    49	    ).build_live_snapshot(receipt_id="inv-r1")
    50	
    51	    assert snapshot.live_ids == ["pos-1"]
    52	    assert snapshot.position_states[0].position_id == "pos-1"
    53	    assert snapshot.position_states[0].source_payload["metadata"] == {"source": "fill"}
    54	    assert snapshot.capital_state is not None
    55	    assert snapshot.capital_state.snapshot_id == "cap-1"
    56	    assert snapshot.exposure_state.position_count == 1
    57	    assert snapshot.exposure_state.declared_exposure_by_symbol == {"NVDA": 12.5}
    58	    assert snapshot.inventory_receipt is not None
    59	    assert snapshot.inventory_receipt.position_ids == ["pos-1"]
    60	    assert snapshot.inventory_receipt.capital_snapshot_id == "cap-1"
    61	
    62	    snapshot.position_states[0].source_payload["metadata"]["source"] = "mutated"
    63	    assert positions.read_position("pos-1") == {
    64	        "symbol": "NVDA",
    65	        "declared_exposure": 12.5,
    66	        "metadata": {"source": "fill"},
    67	    }

    99	def test_live_inventory_module_does_not_import_broker_or_candidate_authority() -> None:
   100	    source = Path("src/nvda_desk/downstream/live_inventory.py").read_text(encoding="utf-8")
   101	    tree = ast.parse(source)
   102	    imported = set()
   103	    for node in ast.walk(tree):
   104	        if isinstance(node, ast.Import):
   105	            imported.update(alias.name for alias in node.names)
   106	        elif isinstance(node, ast.ImportFrom) and node.module:
   107	            imported.add(node.module)
   108	
   109	    forbidden_roots = (
   110	        "nvda_desk.adapters.egress",
   111	        "nvda_desk.domain.execution",
   112	        "nvda_desk.cognition.serial",
   113	        "nvda_desk.ingress.raw_bundle_loader",
   114	    )
   115	    assert not any(
   116	        module == root or module.startswith(root + ".")
   117	        for module in imported
   118	        for root in forbidden_roots
   119	    )
