PROHIBITED_DONOR_IMPORT_PREFIXES = (
    "legacy_import",
    "nvda_desk.schemas.cognition",
    "nvda_desk.schemas.execution",
    "nvda_desk.services.imported_market_context_synthesis",
)

    "ImportedMarketContextSynthesis",
    "LegacyCognitionExtraction",
    "LegacyExecutionRecordShape",
    "RuntimeExecutionDecisionBundle",
}

    "broker_order_id",
    "capital_deployment_authority",
    "open_position",
    "order_placement",
    "position_sizing",
}


def _imports_for(path: Path) -> set[str]:
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    imports: set[str] = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            imports.update(alias.name for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            imports.add(node.module)
    return imports


def _top_level_defs_for(path: Path) -> set[str]:
    tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    return {
        node.name
        for node in tree.body
        if isinstance(node, ast.ClassDef | ast.FunctionDef | ast.AsyncFunctionDef)
    }


def _flatten_terms(value: object) -> Iterable[str]:
    if isinstance(value, dict):
        for key, child in value.items():
            yield str(key)
            yield from _flatten_terms(child)
    elif isinstance(value, (list, tuple, set)):
        for child in value:
            yield from _flatten_terms(child)
    elif value is not None:
        yield str(value)


def test_active_serial_files_do_not_import_donor_cognition_or_execution_stacks() -> None:
    for path in ACTIVE_SERIAL_FILES:
        imports = _imports_for(path)
        forbidden = {
            imported
            for imported in imports
            if any(
                imported == prefix or imported.startswith(f"{prefix}.")
                for prefix in PROHIBITED_DONOR_IMPORT_PREFIXES
            )
        }
        assert forbidden == set(), f"{path} imports donor serial stack modules: {sorted(forbidden)}"


def test_active_serial_files_do_not_redeclare_transform_required_donor_symbols() -> None:
    declared: set[str] = set()
    for path in ACTIVE_SERIAL_FILES:
        declared.update(_top_level_defs_for(path))

    forbidden = declared & PROHIBITED_DONOR_SYMBOLS
    assert forbidden == set(), f"serial native files redeclare donor symbols: {sorted(forbidden)}"


def test_serial_lifecycle_plan_remains_candidate_material_not_opening_or_capital_payload() -> None:
    plan = build_lifecycle_plan(
        ExecutionExpressionOutput(
            lead_playbook_id="playbook::demo",
            active_playbook_ids=["playbook::demo"],
        )
    )

    emitted_terms = set(_flatten_terms(plan))
    assert emitted_terms.isdisjoint(PROHIBITED_DOWNSTREAM_LIFECYCLE_TERMS)

def _require_module(module_path: str):
    try:
        return importlib.import_module(module_path)
    except ModuleNotFoundError as exc:
        pytest.fail(f"Required module is not implemented: {module_path} ({exc})")


def _require_attr(module, attr_name: str):
    if not hasattr(module, attr_name):
        pytest.fail(f"Required attribute is not implemented: {module.__name__}.{attr_name}")
    return getattr(module, attr_name)


def test_posture_contract_and_service_surfaces_are_present() -> None:
    contracts = _require_module("nvda_desk.contracts.posture")
    serial_module = _require_module("nvda_desk.cognition.serial.posture")
    service_module = _require_module("nvda_desk.services.posture_service")

    _require_attr(contracts, "PermissionState")
    _require_attr(contracts, "PosturePermissionEnvelope")
    output_model = _require_attr(contracts, "PostureRiskOutput")
    _require_attr(serial_module, "evaluate_posture")
    _require_attr(service_module, "PostureService")

    assert hasattr(output_model, "model_fields")


def test_posture_output_keeps_permission_envelope_separate_from_capital_decision() -> None:
    contracts = _require_module("nvda_desk.contracts.posture")

def test_posture_service_applies_promoted_coefficient_authority_and_bounded_modifiers() -> None:

    assert (

            "baseline_value"
        ]
        == 0.35
    )
    assert (
        stressed_output.local_envelope["effective_coefficients"]["max_risk_per_trade"][
            "effective_value"
        ]
        == 0.25
    )
    assert stressed_output.local_envelope["effective_coefficients"]["max_risk_per_trade"][
        "modifier_inputs"
    ] == ["negative_gamma_stress"]
    assert (
        stressed_output.local_envelope["effective_coefficients"]["hedge_required"][

    "before_independent_lane",
    "pre_capital_deployment",
    "no_live_inventory_authority",
)


def validate_serial_stage_order(stage_order: Sequence[str]) -> tuple[str, ...]:
    """Return the stage order after rejecting missing, duplicate, or reordered stages."""

    observed = tuple(stage_order)
    if len(set(observed)) != len(observed):
        raise ValueError("serial stage order contains duplicate stages")
    expected = SERIAL_SPINE_STAGE_ORDER
    if observed != expected:
        raise ValueError(
            "serial stage order must be " + ",".join(expected) + "; observed " + ",".join(observed)
        )
    return observed


def validate_serial_stage_dependencies(
    stage_order: Sequence[str], dependencies: Mapping[str, Sequence[str]]
) -> dict[str, tuple[str, ...]]:
    """Validate that stage dependencies are known and never point forward."""

    ordered = validate_serial_stage_order(stage_order)
    dependency_keys = set(dependencies)
    expected_keys = set(ordered)
    if dependency_keys != expected_keys:
        missing = sorted(expected_keys - dependency_keys)
        extra = sorted(dependency_keys - expected_keys)
        raise ValueError(f"serial dependencies mismatch missing={missing} extra={extra}")

    index = {stage_name: pos for pos, stage_name in enumerate(ordered)}
    normalised: dict[str, tuple[str, ...]] = {}
    for stage_name in ordered:
        stage_dependencies = tuple(dependencies[stage_name])
        unknown = [name for name in stage_dependencies if name not in index]
        if unknown:
            raise ValueError(f"serial stage {stage_name} has unknown dependencies {unknown}")
        forward = [name for name in stage_dependencies if index[name] >= index[stage_name]]
        if forward:
            raise ValueError(f"serial stage {stage_name} has forward dependencies {forward}")
        normalised[stage_name] = stage_dependencies
    return normalised


def serial_spine_contract() -> dict[str, Any]:
    """Return a copy-safe trace contract for the serial candidate-formation spine."""

    dependencies = validate_serial_stage_dependencies(
        SERIAL_SPINE_STAGE_ORDER, SERIAL_STAGE_DEPENDENCIES
    )
    return {
        "architecture_row": "5.7 Serial desk-cognition spine",
        "owner": "serial desk-cognition authority",
        "purpose": "ordered candidate formation through six serial runtime stages",
        "stage_order": list(SERIAL_SPINE_STAGE_ORDER),
        "stage_dependencies": {
            stage_name: list(stage_dependencies)
            for stage_name, stage_dependencies in dependencies.items()
        },
        "stage_output_schemas": dict(SERIAL_STAGE_OUTPUT_SCHEMAS),
        "lineage_tags": list(SERIAL_STAGE_LINEAGE_TAGS),
        "prohibited_authorities": list(SERIAL_SPINE_PROHIBITED_AUTHORITIES),
        "boundary": "forms candidate material only; does not allocate capital, place orders, or manage live inventory",
    }


def copy_serial_spine_contract() -> dict[str, Any]:
    """Return a defensive copy of the serial spine trace contract."""

    return deepcopy(serial_spine_contract())

from __future__ import annotations

from datetime import date, datetime

from pydantic import BaseModel, ConfigDict, Field, model_validator


class SourceTemporalStandard(BaseModel):
    """Canonical source temporal carrier admitted above packet emission time."""

    model_config = ConfigDict(extra="forbid")

    source_time_raw: str | None = None
    source_time_kind: str | None = None
    source_timezone_raw: str | None = None
    source_time_utc: datetime | None = None
    source_date: date | None = None
    captured_at_utc: datetime | None = None
    persisted_at_utc: datetime | None = None
    source_provider: str | None = None
    source_channel: str | None = None
    time_granularity: str | None = None

    @model_validator(mode="after")
    def _require_source_temporal_truth(self) -> "SourceTemporalStandard":
        if self.source_time_utc is None and self.source_date is None:
            raise ValueError("source temporal standard requires source_time_utc or source_date")
        return self


class TemporalEventWindow(BaseModel):
    """One bounded temporal event window."""

    model_config = ConfigDict(extra="forbid")

    event_id: str
    window_start_utc: datetime
    window_end_utc: datetime

class TemporalContextOutput(SourceTemporalStandard):

    active_tags: list[str] = Field(default_factory=list)

def build_source_temporal_payload(
    *,
    source_time_utc: datetime | None = None,
    source_date: date | None = None,
    source_time_raw: str | None = None,
    source_time_kind: str | None = None,
    source_timezone_raw: str | None = None,
    captured_at_utc: datetime | None = None,
    persisted_at_utc: datetime | None = None,
    source_provider: str | None = None,
    source_channel: str | None = None,
    time_granularity: str | None = None,

    """Return one canonical source temporal payload mapping."""

    if source_time_kind is None:
        if source_time_utc is not None:

    if time_granularity is None:
        if source_time_utc is not None:
            time_granularity = "timestamp"
        elif source_date is not None:
            time_granularity = "date"
    return {
        "source_time_raw": source_time_raw,
        "source_time_kind": source_time_kind,
        "source_timezone_raw": source_timezone_raw,
        "source_time_utc": source_time_utc,
        "source_date": source_date,
        "captured_at_utc": captured_at_utc,
        "persisted_at_utc": persisted_at_utc,
        "source_provider": source_provider,
        "source_channel": source_channel,

from __future__ import annotations

import hashlib
import json

from typing import Any

from pydantic import ValidationError

from nvda_desk.contracts.downstream import DownstreamLiveInventorySnapshot

from nvda_desk.contracts.execution import ExecutionExpressionInput, StageLocalHandoffSurface
from nvda_desk.contracts.serial_spine import (

from nvda_desk.contracts.stage_inputs import StageInputBundle

class DeskRuntimeOrchestrator:
    """Own one deterministic stateless runtime pass and explicit serial stage order."""

    stage_order: tuple[str, ...] = SERIAL_SPINE_STAGE_ORDER

    _stage_dependencies: dict[str, tuple[str, ...]] = validate_serial_stage_dependencies(
        SERIAL_SPINE_STAGE_ORDER, SERIAL_STAGE_DEPENDENCIES
    )

    _stage_packet_metadata: dict[str, dict[str, str]] = {
        "temporal": {
            "module_id": "temporal_context",
            "grammar_role": "temporal_context",
            "dataschema": "TemporalContextOutput",
        },
        "regime": {
            "module_id": "market_regime_context",
            "grammar_role": "market_regime_context",
            "dataschema": "MarketRegimeContextOutput",
        },
        "options_flow": {
            "module_id": "options_flow_context",
            "grammar_role": "options_flow_context",
            "dataschema": "OptionsFlowContextOutput",
        },
        "posture": {
            "module_id": "posture_risk_permission",
            "grammar_role": "posture_risk_permission",
            "dataschema": "PostureRiskOutput",
        },
        "eligibility": {
            "module_id": "playbook_eligibility",
            "grammar_role": "playbook_eligibility",
            "dataschema": "PlaybookEligibilityOutput",
        },
        "execution": {
            "module_id": "expression_execution",
            "grammar_role": "expression_execution",
            "dataschema": "ExecutionExpressionOutput",
        },
    }

    def stage_sequence(self) -> tuple[str, ...]:
        """Return the governed six-stage serial-runtime order for one deterministic pass."""

        return self.stage_order

    def run_once(
        self,
        stage_inputs: Mapping[str, Any] | StageInputBundle,
        *,
        live_inventory_service: Any | None = None,
        live_inventory_request: Mapping[str, Any] | None = None,
        live_inventory_snapshot: DownstreamLiveInventorySnapshot | Mapping[str, Any] | None = None,
    ) -> dict[str, Any]:
        """Return one bounded runtime pass from the admitted stage-input boundary."""

        bundle = _coerce_admitted_bundle(stage_inputs)
        if bundle is None:
            normalised_stage_inputs = _normalise_stage_inputs(stage_inputs)
            executed = [name for name in self.stage_order if name in normalised_stage_inputs]
            receipt = {
                "stage_order": list(self.stage_order),
                "executed_stages": executed,
                "missing_stages": [
                    name for name in self.stage_order if name not in normalised_stage_inputs
                ],
            }
            return _attach_runtime_live_inventory(
                receipt,
                live_inventory_service=live_inventory_service,
                live_inventory_request=live_inventory_request,
                live_inventory_snapshot=live_inventory_snapshot,
            )
        receipt = self._run_serial_bundle(bundle)
        return _attach_runtime_live_inventory(
            receipt,
            live_inventory_service=live_inventory_service,
            live_inventory_request=live_inventory_request,
            live_inventory_snapshot=live_inventory_snapshot,
        )

    def _run_serial_bundle(self, stage_input_bundle: StageInputBundle) -> dict[str, Any]:
        source_time_utc = _require_source_time_utc(stage_input_bundle)
        token = _time_token(source_time_utc)
        traceid = f"trace::serial_runtime::{token}"
        runid = f"run::serial_runtime::{token}"
        scenarioid = "live"
        inputfingerprint = _input_fingerprint(stage_input_bundle)
        packet_emitted_at = datetime.now(UTC)

        temporal = TemporalContextService().evaluate(stage_input_bundle.temporal)
        regime = RegimeContextService().evaluate(stage_input_bundle.regime)
        options_flow = OptionsFlowContextService().evaluate(stage_input_bundle.options_flow)
        posture = PostureService().evaluate(regime=regime)
        eligibility = PlaybookEligibilityService().evaluate(
            posture=posture, candidate_family_ids=[]
        )
        execution_input = ExecutionExpressionInput(
            temporal=temporal.model_dump(mode="json"),
            regime=regime.model_dump(mode="json"),
            options_flow=options_flow.model_dump(mode="json"),
            posture=posture.model_dump(mode="json"),
            eligibility=eligibility.model_dump(mode="json"),
        )
        execution = evaluate_execution_expression(
            execution_input=execution_input,
            eligibility_output=eligibility,
        )

        stage_outputs = {
            "temporal": temporal,
            "regime": regime,
            "options_flow": options_flow,
            "posture": posture,
            "eligibility": eligibility,
            "execution": execution,
        }
        stage_packets = self._build_stage_packets(
            stage_outputs=stage_outputs,
            source_time_utc=source_time_utc,
            packet_emitted_at=packet_emitted_at,
            traceid=traceid,
            runid=runid,
            scenarioid=scenarioid,
            inputfingerprint=inputfingerprint,
        )

        stage_local_handoff = StageLocalHandoffSurface(
            execution_post_modifier_pre_final_risk=execution.model_dump(mode="json"),
            lineage_tags=list(SERIAL_STAGE_LINEAGE_TAGS),
        )

        return {
            "stage_order": list(self.stage_order),
            "executed_stages": list(self.stage_order),
            "missing_stages": [],
            "serial_spine_contract": copy_serial_spine_contract(),
            "stage_dependencies": {
                stage_name: list(self._stage_dependencies[stage_name])
                for stage_name in self.stage_order
            },
            "stage_packet_ids": {
                stage_name: packet.packet_id for stage_name, packet in stage_packets.items()
            },
            "stage_outputs": {
                stage_name: output.model_dump(mode="json")
                for stage_name, output in stage_outputs.items()
            },
            "stage_packets": {
                stage_name: packet.model_dump(mode="json")
                for stage_name, packet in stage_packets.items()
            },
            "stage_local_handoff": stage_local_handoff.model_dump(mode="json"),
        }

    def _build_stage_packets(
        self,
        *,
        stage_outputs: dict[str, Any],
        source_time_utc: datetime,
        packet_emitted_at: datetime,
        traceid: str,
        runid: str,
        scenarioid: str,
        inputfingerprint: str,
    ) -> dict[str, Any]:
        stage_packet_ids = {
            stage_name: f"dmp::serial::{stage_name}::{_time_token(source_time_utc)}"
            for stage_name in self.stage_order
        }
        packets: dict[str, Any] = {}
        previous_packet_id: str | None = None
        for stage_name in self.stage_order:
            metadata = self._stage_packet_metadata[stage_name]
            dependencyids = [
                stage_packet_ids[name] for name in self._stage_dependencies[stage_name]
            ]
            packet = build_stage_packet(
                packet_id=stage_packet_ids[stage_name],
                dataschema=metadata["dataschema"],
                data=stage_outputs[stage_name],
                emitted_at=packet_emitted_at,
                traceid=traceid,
                runid=runid,
                scenarioid=scenarioid,
                module_id=metadata["module_id"],
                module_version="native",
                module_instance_id=f"{metadata['module_id']}::runtime",
                grammar_role=metadata["grammar_role"],
                stage_name=stage_name,
                parentids=[previous_packet_id] if previous_packet_id is not None else [],
                dependencyids=dependencyids,
                derivedfrom=_derived_from(stage_name, self._stage_dependencies[stage_name]),
                sourceartifactids=[],
                inputfingerprint=inputfingerprint,
                summary_headline=_stage_summary(stage_name, stage_outputs[stage_name]),
                dmpextensions={
                    "nvda": {
                        "schema_identifiers": {
                            "output_model_name": metadata["dataschema"],
                        },
                    }
                },
            )
            packets[stage_name] = packet
            previous_packet_id = packet.packet_id
        return packets


def _attach_runtime_live_inventory(
    receipt: dict[str, Any],
    *,
    live_inventory_service: Any | None,
    live_inventory_request: Mapping[str, Any] | None,
    live_inventory_snapshot: DownstreamLiveInventorySnapshot | Mapping[str, Any] | None,
) -> dict[str, Any]:
    if live_inventory_service is not None and live_inventory_snapshot is not None:
        raise ValueError("live_inventory_service and live_inventory_snapshot cannot both be supplied")
    if live_inventory_service is None and live_inventory_snapshot is None:
        if live_inventory_request is not None:
            raise ValueError("live_inventory_request requires live_inventory_service")
        return receipt
    if live_inventory_snapshot is not None:
        snapshot = _coerce_live_inventory_snapshot(live_inventory_snapshot)
    else:
        request = dict(live_inventory_request or {})
        allowed_keys = {"live_ids", "inventory_id", "capital_snapshot_id", "receipt_id"}
        extra_keys = set(request) - allowed_keys
        if extra_keys:
            joined = ", ".join(sorted(extra_keys))
            raise ValueError(f"live_inventory_request contains unsupported keys: {joined}")
        snapshot = live_inventory_service.build_live_snapshot(**request)
    receipt["runtime_live_inventory"] = snapshot.model_dump(mode="json")
    receipt["runtime_live_inventory_policy_effect"] = (
        "read_only_no_serial_decision_change"
    )
    return receipt


def _coerce_live_inventory_snapshot(
    live_inventory_snapshot: DownstreamLiveInventorySnapshot | Mapping[str, Any],
) -> DownstreamLiveInventorySnapshot:
    if isinstance(live_inventory_snapshot, DownstreamLiveInventorySnapshot):
        return live_inventory_snapshot
    return DownstreamLiveInventorySnapshot.model_validate(live_inventory_snapshot)


def _coerce_admitted_bundle(
    stage_inputs: Mapping[str, Any] | StageInputBundle,
) -> StageInputBundle | None:
    if isinstance(stage_inputs, StageInputBundle):
        return stage_inputs
    if isinstance(stage_inputs, Mapping):
        try:
            return StageInputBundle.model_validate(stage_inputs)
        except ValidationError:
            return None
    return None


def _normalise_stage_inputs(
    stage_inputs: Mapping[str, Any] | StageInputBundle,
) -> Mapping[str, Any]:
    if isinstance(stage_inputs, StageInputBundle):
        return stage_inputs.as_mapping()
    return stage_inputs


def _require_source_time_utc(stage_input_bundle: StageInputBundle) -> datetime:
    source_time_utc = stage_input_bundle.temporal.source_time_utc
    if source_time_utc is None:
        raise ValueError("Serial runtime requires temporal.source_time_utc.")
    if source_time_utc.tzinfo is None:
        return source_time_utc
    return source_time_utc.astimezone(UTC)


def _time_token(source_time_utc: datetime) -> str:
    return source_time_utc.isoformat().replace("+00:00", "Z").replace(":", "").replace("-", "")


def _input_fingerprint(stage_input_bundle: StageInputBundle) -> str:
    payload = json.dumps(
        stage_input_bundle.model_dump(mode="json"), sort_keys=True, separators=(",", ":")
    )
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()


def _derived_from(stage_name: str, dependencies: tuple[str, ...]) -> str:
    if not dependencies:
        return f"{stage_name}_from_declared_stage_inputs"
    dependency_label = "+".join(dependencies)
    return f"{stage_name}_from_{dependency_label}"


def _stage_summary(stage_name: str, payload: Any) -> str:
    if stage_name == "temporal":
        return f"session_state={payload.session_state}"
    if stage_name == "regime":
        return f"volatility_regime={payload.volatility_regime}"
    if stage_name == "options_flow":
        return f"term_structure_state={payload.term_structure_state}"
    if stage_name == "posture":
        return f"permission_state={payload.permission_state.value}"
    if stage_name == "eligibility":
        return f"active_families={','.join(payload.active_family_ids) if payload.active_family_ids else 'none'}"
    if stage_name == "execution":
        return f"lead_playbook_id={payload.lead_playbook_id or 'none'}"
    raise ValueError(f"Unknown stage name: {stage_name}")

"""Canonical DMPV3 packet-carriage contracts for the live native repo."""

from __future__ import annotations

from copy import deepcopy
from datetime import UTC, datetime
from typing import Any, Generic, TypeVar

from pydantic import BaseModel, ConfigDict, Field

T = TypeVar("T")


class DmpV3Producer(BaseModel):
    """Bounded producer identity carried by one DMPV3 packet."""

    model_config = ConfigDict(extra="forbid")

    module_id: str = Field(min_length=1)
    module_version: str = Field(min_length=1)
    module_instance_id: str = Field(min_length=1)
    grammar_role: str = Field(min_length=1)
    behaviour_class: str = Field(min_length=1)
    stage_name: str | None = None


class DmpV3Lineage(BaseModel):
    """Bounded lineage surface for one DMPV3 packet."""

    model_config = ConfigDict(extra="forbid")

    parentids: list[str] = Field(default_factory=list)
    dependencyids: list[str] = Field(default_factory=list)
    derivedfrom: str | None = None
    replayref: str | None = None
    sourceartifactids: list[str] = Field(default_factory=list)
    inputfingerprint: str | None = None
    reviewtraceid: str | None = None
    replaytraceid: str | None = None
    decisiontraceid: str | None = None


class DmpV3ValidationCheck(BaseModel):
    """One named validation check outcome."""

    model_config = ConfigDict(extra="forbid")

    name: str = Field(min_length=1)
    passed: bool


class DmpV3Validation(BaseModel):
    """Validation carriage for one DMPV3 packet."""

    model_config = ConfigDict(extra="forbid")

    status: str = Field(default="valid", min_length=1)
    checks: list[DmpV3ValidationCheck] = Field(
        default_factory=lambda: [DmpV3ValidationCheck(name="schema_validation", passed=True)]
    )
    violations: list[str] = Field(default_factory=list)


class DmpV3Summary(BaseModel):
    """Human-facing summary block carried by one DMPV3 packet."""

    model_config = ConfigDict(extra="forbid")

    headline: str = Field(min_length=1)
    keyflags: list[str] = Field(default_factory=list)
    humanstate: str = Field(min_length=1)


def _deep_merge_dicts(base: dict[str, Any], updates: dict[str, Any]) -> dict[str, Any]:
    """Recursively merge bounded extension dictionaries without erasing canonical leaves."""

    merged = deepcopy(base)
    for key, value in updates.items():
        if isinstance(value, dict) and isinstance(merged.get(key), dict):
            merged[key] = _deep_merge_dicts(merged[key], value)
        else:
            merged[key] = deepcopy(value)
    return merged


class DmpV3Envelope(BaseModel, Generic[T]):
    """Canonical DMPV3 outer envelope for live inter-module packet carriage."""

    model_config = ConfigDict(extra="forbid")

    specversion: str = "1.0"
    dmpversion: str = "3.0.0"
    id: str = Field(min_length=1)
    source: str = Field(min_length=1)
    type: str = Field(min_length=1)
    time: datetime
    datacontenttype: str = "application/json"
    dataschema: str = Field(min_length=1)
    traceid: str = Field(min_length=1)
    runid: str = Field(min_length=1)
    scenarioid: str = Field(min_length=1)
    packetrole: str = Field(min_length=1)
    producer: DmpV3Producer
    lineage: DmpV3Lineage = Field(default_factory=DmpV3Lineage)
    validation: DmpV3Validation = Field(default_factory=DmpV3Validation)
    summary: DmpV3Summary
    data: T
    dmpextensions: dict[str, Any] = Field(default_factory=dict)

    @property
    def packet_id(self) -> str:
        return self.id

    @property

        return self.time

def build_dmp_v3_envelope_kwargs(
    *,
    packet_id: str,
    dataschema: str,
    data: Any,
    packetrole: str,
    module_id: str,
    module_version: str = "native",
    module_instance_id: str | None = None,
    grammar_role: str,
    behaviour_class: str,
    stage_name: str | None = None,
    emitted_at: datetime | None = None,
    source: str | None = None,
    packet_type: str | None = None,
    datacontenttype: str = "application/json",
    traceid: str | None = None,
    runid: str | None = None,
    scenarioid: str | None = None,
    parentids: list[str] | None = None,
    dependencyids: list[str] | None = None,
    derivedfrom: str | None = None,
    replayref: str | None = None,
    sourceartifactids: list[str] | None = None,
    inputfingerprint: str | None = None,
    reviewtraceid: str | None = None,
    replaytraceid: str | None = None,
    decisiontraceid: str | None = None,
    validation_violations: list[str] | None = None,
    summary_headline: str | None = None,
    summary_keyflags: list[str] | None = None,
    summary_humanstate: str | None = None,
    dmpextensions: dict[str, Any] | None = None,
) -> dict[str, Any]:
    """Return canonical DMPV3 envelope kwargs for one repo-owned payload."""

    if emitted_at is None:
        emitted_at = datetime.now(UTC)

    violations = list(validation_violations or [])
    resolved_traceid = traceid
    if resolved_traceid is None:
        resolved_traceid = f"trace::{packet_id}"
        violations.append("traceid_defaulted_from_packet_id")

    resolved_runid = runid
    if resolved_runid is None:
        resolved_runid = f"run::{packet_id}"
        violations.append("runid_defaulted_from_packet_id")

    resolved_scenarioid = scenarioid or "live"
    resolved_source = source or f"urn:nvda:producer:{module_id}"
    resolved_type = packet_type or f"urn:nvda:packet:{packetrole}"

    lineage = DmpV3Lineage(
        parentids=list(parentids or []),
        dependencyids=list(dependencyids or []),
        derivedfrom=derivedfrom,
        replayref=replayref,
        sourceartifactids=list(sourceartifactids or []),
        inputfingerprint=inputfingerprint,
        reviewtraceid=reviewtraceid,
        replaytraceid=replaytraceid,
        decisiontraceid=decisiontraceid,
    )

    validation = DmpV3Validation(violations=violations)
    if violations:
        validation.status = "degraded"
        validation.checks = [DmpV3ValidationCheck(name="carriage_context", passed=False)]

    payload_type = data.__class__
    canonical_schema_identifiers = {
        "payload_model_name": payload_type.__name__,
        "payload_module_path": payload_type.__module__,
        "output_model_name": payload_type.__name__,
    }
    extensions_payload: dict[str, Any] = {
        "nvda": {
            "schema_identifiers": canonical_schema_identifiers,
            "semantic_dependencies": list(lineage.dependencyids),
        }
    }
    if dmpextensions:
        extensions_payload = _deep_merge_dicts(extensions_payload, dict(dmpextensions))
        canonical_nvda_extensions = extensions_payload.setdefault("nvda", {})
        canonical_nvda_extensions["schema_identifiers"] = canonical_schema_identifiers
        canonical_nvda_extensions["semantic_dependencies"] = list(lineage.dependencyids)

    keyflags = list(summary_keyflags or [])
    if validation.status == "degraded":
        for violation in violations:
            if violation not in keyflags:
                keyflags.append(violation)

    summary = DmpV3Summary(
        headline=summary_headline or f"{module_id} emitted {packetrole}",
        keyflags=keyflags,
        humanstate=summary_humanstate or validation.status,
    )

    return {
        "id": packet_id,
        "source": resolved_source,
        "type": resolved_type,
        "time": emitted_at,
        "datacontenttype": datacontenttype,
        "dataschema": dataschema,
        "traceid": resolved_traceid,
        "runid": resolved_runid,
        "scenarioid": resolved_scenarioid,
        "packetrole": packetrole,
        "producer": DmpV3Producer(
            module_id=module_id,
            module_version=module_version,
            module_instance_id=module_instance_id or f"{module_id}::default",
            grammar_role=grammar_role,
            behaviour_class=behaviour_class,
            stage_name=stage_name,
        ),
        "lineage": lineage,
        "validation": validation,
        "summary": summary,
        "data": data,
        "dmpextensions": extensions_payload,
    }
